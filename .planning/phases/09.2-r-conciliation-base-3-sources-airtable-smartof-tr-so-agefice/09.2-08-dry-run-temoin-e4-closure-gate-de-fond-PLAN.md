Selection session temoin (SQL read-only via docker exec qualiof_postgres psql -U qualiof -d qualiof) — colonnes CNI/RIB = SensitiveData.cniKey/ribKey (et Person.ribKey) :
SELECT s.code, s.name, s."startDate", count(sp.id) AS nb_part,
  count(sd."personId") FILTER (WHERE sd."cniKey" IS NOT NULL AND sd."ribKey" IS NOT NULL) AS nb_cni_rib,
  count(ap.id) FILTER (WHERE ap.id IS NOT NULL) AS nb_agefice, p."durationHours"
FROM "TrainingSession" s
JOIN "SessionParticipant" sp ON sp."sessionId"=s.id
JOIN "Organization" o ON o.id=sp."sponsorOrgId" AND o."opcoCode"='AGEFICE'
JOIN "Person" per ON per.id=sp."personId"
LEFT JOIN "SensitiveData" sd ON sd."personId"=per.id
LEFT JOIN "AgeficeProfile" ap ON ap."organizationId"=o.id
JOIN "TrainingProduct" p ON p.id=s."productId"
WHERE s."startDate">='2026-01-01' AND s.status<>'CANCELLED'
GROUP BY s.code,s.name,s."startDate",p."durationHours"
HAVING count(sp.id)>=2
ORDER BY p."durationHours" DESC, nb_cni_rib DESC;
(Adapter les noms de colonnes si le schema differe ; profil cible Dumoulin/Barriere echantillon BCI.)

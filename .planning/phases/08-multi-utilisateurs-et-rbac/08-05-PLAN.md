    Étape 1 — Créer `apps/web/src/lib/build-audit-where.ts` :

    ```typescript
    import type { Prisma } from '@prisma/client';

    export interface AuditFilters {
      userId?: string;
      action?: string;
      from?: string;
      to?: string;
    }

    function parseDate(s: string | undefined): Date | undefined {
      if (!s) return undefined;
      const d = new Date(s);
      return Number.isNaN(d.getTime()) ? undefined : d;
    }

    export function buildAuditWhere(filters: AuditFilters, tenantId: string): Prisma.AuditLogWhereInput {
      const where: Prisma.AuditLogWhereInput = { tenantId };
      if (filters.userId) where.userId = filters.userId;
      if (filters.action) where.action = { startsWith: filters.action };
      const from = parseDate(filters.from);
      const to = parseDate(filters.to);
      if (from || to) {
        where.createdAt = {};
        if (from) where.createdAt.gte = from;
        if (to) where.createdAt.lte = to;
      }
      return where;
    }
    ```

    ```typescript
    'use server';
    import { cookies } from 'next/headers';
    import { redirect } from 'next/navigation';
    import argon2 from 'argon2';
    import { prisma } from '@qualiof/db';
    import { loginSchema, type LoginInput } from '@qualiof/shared';
    import { lucia } from '@/lib/auth';

    async function safeAudit(data: {
      tenantId: string | null;
      userId: string | null;
      targetUserId: string;
      action: string;
      diff?: Record<string, unknown>;
    }) {
      try {
        if (!data.tenantId) return;  // pas de tenant identifié → ne pas logger (FK obligatoire)
        await prisma.auditLog.create({
          data: {
            tenantId: data.tenantId,
            userId: data.userId,
            entity: 'User',
            entityId: data.targetUserId,
            action: data.action,
            diff: (data.diff ?? {}) as never,
          },
        });
      } catch {
        // best-effort : audit log ne doit JAMAIS bloquer le login
      }
    }

    export async function loginAction(input: LoginInput): Promise<{ ok: true } | { ok: false; error: string }> {
      const parsed = loginSchema.safeParse(input);
      if (!parsed.success) return { ok: false, error: 'Données invalides.' };
      const { email, password } = parsed.data;
      const lcEmail = email.toLowerCase();

      const user = await prisma.user.findUnique({ where: { email: lcEmail } });
      if (!user) {
        // pas de tenantId → pas d'AuditLog persistable (FK contrainte). On skip.
        return { ok: false, error: 'Identifiants invalides.' };
      }

      if (user.disabledAt) {
        await safeAudit({ tenantId: user.tenantId, userId: user.id, targetUserId: user.id,
          action: 'auth.login.failed', diff: { email: lcEmail, reason: 'disabled' } });
        return { ok: false, error: 'Compte désactivé. Contactez votre administrateur.' };
      }

      const ok = await argon2.verify(user.hashedPwd, password);
      if (!ok) {
        await safeAudit({ tenantId: user.tenantId, userId: user.id, targetUserId: user.id,
          action: 'auth.login.failed', diff: { email: lcEmail, reason: 'bad_password' } });
        return { ok: false, error: 'Identifiants invalides.' };
      }

      // Succès
      await prisma.user.update({ where: { id: user.id }, data: { lastLoginAt: new Date() } });
      await safeAudit({ tenantId: user.tenantId, userId: user.id, targetUserId: user.id, action: 'auth.login.success' });

      const session = await lucia.createSession(user.id, {});
      const sessionCookie = lucia.createSessionCookie(session.id);
      (await cookies()).set(sessionCookie.name, sessionCookie.value, sessionCookie.attributes);
      redirect('/app');
    }

    export async function logoutAction(): Promise<void> {
      const sessionId = (await cookies()).get(lucia.sessionCookieName)?.value;
      if (sessionId) {
        await lucia.invalidateSession(sessionId);
      }
      const blank = lucia.createBlankSessionCookie();
      (await cookies()).set(blank.name, blank.value, blank.attributes);
      redirect('/login');
    }
    ```

    ```typescript
      if (sp.to) qs.set('to', sp.to);
      const prevQs = new URLSearchParams(qs); prevQs.set('page', String(Math.max(0, page - 1)));
      const nextQs = new URLSearchParams(qs); nextQs.set('page', String(page + 1));

      return (
        <div className="space-y-6">
          <div>
            <h1 className="text-2xl font-bold">Historique</h1>

            <table className="w-full text-sm">

                <tr>
                  <th className="text-left p-3">Date</th>
                  <th className="text-left p-3">Utilisateur</th>
                  <th className="text-left p-3">Action</th>
                  <th className="text-left p-3">Entité</th>
                  <th className="text-right p-3">Diff</th>
                </tr>
              </thead>
              <tbody>
                {rows.map((r) => (
                  <tr key={r.id}>
                    <td className="p-3 text-xs">{fmtDate(r.createdAt)}</td>
                    <td className="p-3">{r.user ? `${r.user.firstName} ${r.user.lastName}` : <span className="text-muted-foreground italic">—</span>}</td>
                    <td className="p-3"><span className="text-xs px-1.5 py-0.5 rounded bg-slate-100">{r.action}</span></td>
                    <td className="p-3 text-xs text-muted-foreground">{r.entity}:{r.entityId.slice(0, 8)}</td>
                    <td className="p-3 text-right"><AuditDiffModal diff={r.diff} action={r.action} /></td>
                  </tr>
                ))}
                {rows.length === 0 && (
                  <tr><td colSpan={5} className="p-8 text-center text-muted-foreground text-sm">Aucune action ne correspond aux filtres.</td></tr>
                )}
              </tbody>
            </table>
          </div>
          <div className="flex items-center justify-between text-sm">
            <span className="text-muted-foreground">Page {page + 1} / {totalPages}</span>
            <div className="flex gap-2">
              <Link href={`/app/parametres/historique?${prevQs.toString()}` as Route} className={page === 0 ? 'pointer-events-none opacity-50' : ''}>← Précédent</Link>
              <Link href={`/app/parametres/historique?${nextQs.toString()}` as Route} className={page + 1 >= totalPages ? 'pointer-events-none opacity-50' : ''}>Suivant →</Link>
            </div>
          </div>
        </div>
      );
    }
    ```

    ```typescript
    'use client';
    import { useState } from 'react';
    import * as Dialog from '@radix-ui/react-dialog';

    export function AuditDiffModal({ diff, action }: { diff: unknown; action: string }) {
      const [open, setOpen] = useState(false);
      const isBeforeAfterDiff = (
        typeof diff === 'object' && diff !== null && !Array.isArray(diff) &&
        Object.values(diff as Record<string, unknown>).every(v =>
          typeof v === 'object' && v !== null && 'before' in v && 'after' in v
        ) && Object.keys(diff).length > 0
      );
      return (
        <Dialog.Root open={open} onOpenChange={setOpen}>
          <Dialog.Trigger asChild>
            <button type="button" className="text-xs text-primary hover:underline">Voir</button>
          </Dialog.Trigger>
          <Dialog.Portal>
            <Dialog.Overlay className="fixed inset-0 bg-foreground/40 backdrop-blur-sm z-40" />
            <Dialog.Content className="fixed left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 z-50 w-[640px] max-w-[92vw] max-h-[80vh] overflow-y-auto rounded-lg border border-border bg-white p-6 shadow-xl">
              <Dialog.Title className="text-lg font-semibold mb-1">Détail : {action}</Dialog.Title>
              <Dialog.Description className="text-xs text-muted-foreground mb-4">Modifications enregistrées dans l'AuditLog.</Dialog.Description>
              {isBeforeAfterDiff ? (
                <table className="w-full text-sm">
                  <thead className="text-xs uppercase text-muted-foreground"><tr><th className="text-left py-2">Champ</th><th className="text-left py-2">Avant</th><th className="text-left py-2">Après</th></tr></thead>
                  <tbody>
                    {Object.entries(diff as Record<string, { before: unknown; after: unknown }>).map(([k, v]) => (
                      <tr key={k} className="border-t border-border">
                        <td className="py-2 font-medium">{k}</td>
                        <td className="py-2 text-red-700 line-through">{JSON.stringify(v.before)}</td>
                        <td className="py-2 text-emerald-700">{JSON.stringify(v.after)}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              ) : (
                <pre className="text-xs bg-slate-50 border border-border rounded p-3 overflow-x-auto">{JSON.stringify(diff, null, 2)}</pre>
              )}
              <div className="flex justify-end mt-4">
                <Dialog.Close asChild><button type="button" className="rounded-md border border-border px-3 py-1.5 text-sm">Fermer</button></Dialog.Close>
              </div>
            </Dialog.Content>
          </Dialog.Portal>
        </Dialog.Root>
      );
    }
    ```

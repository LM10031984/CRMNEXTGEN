const THEME_LABELS: Record<string, string> = {
  INDIC_23: 'Indicateur 23 — Veille évolutions du secteur de la formation professionnelle',
  INDIC_24: 'Indicateur 24 — Veille évolutions du secteur d\'activité (immobilier)',
  INDIC_25: 'Indicateur 25 — Veille innovations pédagogiques et technologiques',
  INDIC_26: 'Indicateur 26 — Veille handicap et accessibilité (Agefiph + DREETS)',
};

export interface VeilleAuditData {
  theme: 'INDIC_23' | 'INDIC_24' | 'INDIC_25' | 'INDIC_26';
  themeLabel: string;
  watches: Array<{
    title: string;
    url: string | null;
    source: string | null;
    responsable: string | null;
    frequency: string | null;
    exploitation: string | null;
    dateLastReviewed: Date | null;
    dateAdded: Date;
  }>;
  generatedAt: Date;
  tenantName: string;
}

interface TenantSnapshot {
  siret: string;
  nda: string;
  address?: string;
  logoDataUrl?: string;
}

function fmtDate(d: Date | null): string {
  if (!d) return '—';
  return new Intl.DateTimeFormat('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric' }).format(d);
}

function escapeHtml(s: string | null): string {
  if (!s) return '';
  return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

export function getThemeLabel(theme: VeilleAuditData['theme']): string {
  return THEME_LABELS[theme] ?? theme;
}

export function renderVeilleAuditHtml(data: VeilleAuditData, ofConfigSnapshot: TenantSnapshot): string {
  const rows = data.watches.map((w) => `
    <tr>
      <td>${escapeHtml(w.title)}${w.url ? `<br/><a href="${escapeHtml(w.url)}" style="color:#1d4ed8;font-size:9pt">${escapeHtml(w.url)}</a>` : ''}</td>
      <td>${escapeHtml(w.source)}</td>
      <td>${escapeHtml(w.responsable)}</td>
      <td>${escapeHtml(w.frequency)}</td>
      <td>${fmtDate(w.dateLastReviewed)}</td>
      <td>${escapeHtml(w.exploitation)}</td>
    </tr>
  `).join('\n');

  return `<!DOCTYPE html>
<html lang="fr">
<head>
<meta charset="UTF-8" />
<title>Veille Qualiopi — ${escapeHtml(data.themeLabel)}</title>
<style>
  @page { size: A4; margin: 25mm 20mm 30mm 20mm; }
  body { font-family: 'Helvetica', sans-serif; font-size: 10pt; color: #1f2937; }
  header { border-bottom: 2px solid #1f2937; padding-bottom: 8mm; margin-bottom: 8mm; }
  header .row { display: flex; justify-content: space-between; align-items: center; }
  header img.logo { max-height: 50px; }
  h1 { font-size: 16pt; margin: 4mm 0 2mm 0; }
  h2 { font-size: 12pt; color: #4b5563; font-weight: normal; }
  table { width: 100%; border-collapse: collapse; }
  th, td { border: 1px solid #d1d5db; padding: 6px 8px; vertical-align: top; font-size: 9pt; }
  th { background: #f3f4f6; text-align: left; font-weight: bold; }
  tr { page-break-inside: avoid; }
  .meta { font-size: 9pt; color: #6b7280; margin-top: 6mm; }
  ${renderOfPagedFooter({
    tenantName: data.tenantName,
    siret: ofConfigSnapshot.siret,
    nda: ofConfigSnapshot.nda,
    address: ofConfigSnapshot.address,
  })}
</style>
</head>
<body>
  <header>
    <div class="row">
      <div>
        ${ofConfigSnapshot.logoDataUrl ? `<img class="logo" src="${ofConfigSnapshot.logoDataUrl}" alt="${escapeHtml(data.tenantName)}" />` : ''}
      </div>
      <div style="text-align:right;font-size:9pt;color:#6b7280">
        Export Qualiopi — ${fmtDate(data.generatedAt)}
      </div>
    </div>
    <h1>${escapeHtml(data.themeLabel)}</h1>
    <h2>Veille réglementaire — ${escapeHtml(data.tenantName)}</h2>
  </header>

  <main>
    <p class="meta">${data.watches.length} source${data.watches.length > 1 ? 's' : ''} active${data.watches.length > 1 ? 's' : ''} dans cet indicateur.</p>
    <table>
      <thead>
        <tr>
          <th style="width:24%">Source</th>
          <th style="width:12%">Type</th>
          <th style="width:10%">Resp.</th>
          <th style="width:10%">Fréquence</th>
          <th style="width:11%">Dernière revue</th>
          <th style="width:33%">Exploitation</th>

**Étape A — Server action** (`apps/web/src/server/actions/veille-export.ts`)

```typescript
'use server';

import { prisma } from '@qualiof/db';
import { requireRole, ForbiddenError } from '@/lib/rbac';
import { renderVeilleAuditHtml, getThemeLabel } from '@/lib/veille-audit-template';
import { renderHtmlToPdfWeasy } from '@/lib/pdf-render'; // vérifier signature exacte dans pdf-render.ts
import { uploadDocumentPdf } from '@/lib/storage'; // vérifier helper exact ; sinon utiliser uploadToS3
import { resolveOfConfig } from '@/lib/of-config'; // vérifier signature ; peut être getOfConfig(tenantId)
import { logRegulatoryWatchEvent } from '@/lib/regulatoryWatch-audit';
import { createHash } from 'node:crypto';
import { revalidatePath } from 'next/cache';

type Theme = 'INDIC_23' | 'INDIC_24' | 'INDIC_25' | 'INDIC_26';

export async function generateVeilleAuditForTheme(theme: Theme): Promise<
  { ok: true; documentId: string; pdfUrl: string; count: number } | { ok: false; error: string }
> {
  try {
    const user = await requireRole(['ADMIN', 'MANAGER']);

    const [watches, tenant, ofConfig] = await Promise.all([
      prisma.regulatoryWatch.findMany({
        where: { tenantId: user.tenantId, theme, status: 'ACTIVE' },
        orderBy: [{ dateLastReviewed: 'desc' }, { dateAdded: 'desc' }],
      }),
      prisma.tenant.findUnique({ where: { id: user.tenantId } }),
      resolveOfConfig(user.tenantId), // {siret, nda, address, logoDataUrl, name}
    ]);

    if (!tenant) return { ok: false, error: 'Tenant not found' };

    const html = renderVeilleAuditHtml(
      {
        theme,
        themeLabel: getThemeLabel(theme),
        watches: watches.map((w) => ({
          title: w.title,
          url: w.url,
          source: w.source,
          responsable: w.responsable,
          frequency: w.frequency,
          exploitation: w.exploitation,
          dateLastReviewed: w.dateLastReviewed,
          dateAdded: w.dateAdded,
        })),
        generatedAt: new Date(),
        tenantName: ofConfig.name ?? tenant.name ?? 'Organisme de Formation',
      },
      {
        siret: ofConfig.siret ?? tenant.siret ?? '',
        nda: ofConfig.nda ?? tenant.declarationActivite ?? '',
        address: ofConfig.address,
        logoDataUrl: ofConfig.logoDataUrl,
      },
    );

    const pdfBuffer = await renderHtmlToPdfWeasy(html);
    if (!pdfBuffer || pdfBuffer.length < 1000) {
      return { ok: false, error: 'PDF generation failed (buffer too small)' };
    }
    const sha256 = createHash('sha256').update(pdfBuffer).digest('hex');

    // Upload MinIO — D-02
    const upload = await uploadDocumentPdf({
      tenantId: user.tenantId,
      // Convention path : veille-audit/{tenantId}/{theme}-{YYYY-MM-DD-HHmm}.pdf
      keyHint: `veille-audit/${user.tenantId}/${theme}-${new Date().toISOString().replace(/[:T]/g,'-').slice(0,16)}.pdf`,
      content: pdfBuffer,
      contentType: 'application/pdf',
    });
    // upload retourne { key, url } — vérifier signature exacte de storage.ts

    // INSERT Document row — D-02
    const doc = await prisma.document.create({
      data: {
        tenantId: user.tenantId,
        type: 'VEILLE_AUDIT',
        entityType: 'RegulatoryWatch',
        entityId: theme, // ID virtuel = thème (pas une ligne watch précise, c'est un export collectif)
        pdfUrl: upload.url, // ou upload.key selon convention storage
        hashSha256: sha256,
        status: 'generated',
        sessionId: null,
        participantId: null,
      },
    });

    // AuditLog `regulatoryWatch.exported` — D-02 traçabilité
    await logRegulatoryWatchEvent({
      tenantId: user.tenantId,
      actorUserId: user.id,
      targetWatchId: 'BULK',
      action: 'regulatoryWatch.exported',
      diff: { theme, count: watches.length, documentId: doc.id, sha256 },
    });

    revalidatePath('/app/veille');
    return { ok: true, documentId: doc.id, pdfUrl: upload.url, count: watches.length };
  } catch (e) {
    if (e instanceof ForbiddenError) return { ok: false, error: 'Forbidden — ADMIN ou MANAGER requis' };
    console.error('[generateVeilleAuditForTheme] error', e);
    return { ok: false, error: (e as Error).message ?? 'unknown' };
  }
}
```

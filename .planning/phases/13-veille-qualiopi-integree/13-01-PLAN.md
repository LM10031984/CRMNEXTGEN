```typescript
const MONTH_MAP: Record<string, number> = {
  Jan: 0, Feb: 1, Mar: 2, Apr: 3, May: 4, Jun: 5,
  Jul: 6, Aug: 7, Sep: 8, Oct: 9, Nov: 10, Dec: 11,
};

/**
 * Parser de date multi-format utilisé pour l'import du xlsx veille Qualiopi.
 * Tolère 3 formats (DD/MM/YYYY, DD-Mmm-YY, Mmm-YY) + null/empty/unknown → null.
 * Pas d'année implicite (Mmm-YY sans année = pas de match).
 */
export function parseFlexibleDate(v: string | null): Date | null {
  if (!v || !v.trim()) return null;
  const s = v.trim();
  // Format DD/MM/YYYY ou DD/MM/YY
  let m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{2,4})$/);
  if (m) {
    const yy = parseInt(m[3], 10);
    return new Date(yy < 100 ? 2000 + yy : yy, parseInt(m[2], 10) - 1, parseInt(m[1], 10));
  }
  // Format DD-Mmm-YY
  m = s.match(/^(\d{1,2})-([A-Za-z]{3})-(\d{2,4})$/);
  if (m) {
    const yy = parseInt(m[3], 10);
    const mon = MONTH_MAP[m[2]];
    if (mon === undefined) return null;
    return new Date(yy < 100 ? 2000 + yy : yy, mon, parseInt(m[1], 10));
  }
  // Format Mmm-YY (1er du mois par défaut)
  m = s.match(/^([A-Za-z]{3})-(\d{2,4})$/);
  if (m) {
    const yy = parseInt(m[2], 10);
    const mon = MONTH_MAP[m[1]];
    if (mon === undefined) return null;
    return new Date(yy < 100 ? 2000 + yy : yy, mon, 1);
  }
  return null;
}
```

      responsable: normalizeResponsable(normalizeString(r[offset + 4])),
      frequency: normalizeString(r[offset + 5]),
      // Exploitation : col 6 (raw) vs col 8 (exploitation rédigée) sur dual layout
      rawSnippet: hasDualExploitation ? normalizeString(r[offset + 6]) : null,
      exploitation: hasDualExploitation ? normalizeString(r[offset + 8]) : normalizeString(r[offset + 6]),
      dateLastReviewed: parseFlexibleDate(hasDualExploitation ? normalizeString(r[offset + 7]) : null),
    });
  }
  return rows;
}

async function importVeille(tenantId: string, xlsxPath: string): Promise<{ inserted: number; updated: number; skipped: number; total: number }> {
  const buf = fs.readFileSync(xlsxPath); // PAS XLSX.readFile (mémoire)
  const wb = XLSX.read(buf, { type: 'buffer', cellDates: true });

  let inserted = 0, updated = 0, skipped = 0, total = 0;

  for (const { name, theme, headerRow, hasDualExploitation } of SHEET_THEME_MAP) {
    const sheet = wb.Sheets[name];
    if (!sheet) { console.warn(`Sheet absent: ${name}`); continue; }
    const rows = processSheet(sheet, theme, headerRow, hasDualExploitation);
    for (const row of rows) {
      total++;
      // Idempotence : findFirst sur (tenantId, theme, title, url)
      const existing = await prisma.regulatoryWatch.findFirst({
        where: { tenantId, theme, title: row.title, url: row.url },
        select: { id: true },
      });
      if (existing) {
        // UPDATE seulement les champs modifiables (pas de log si pas de changement)
        await prisma.regulatoryWatch.update({
          where: { id: existing.id },
          data: {
            modeSuivi: row.modeSuivi,
            typeSource: row.typeSource,
            responsable: row.responsable,
            frequency: row.frequency,
            exploitation: row.exploitation,
            rawSnippet: row.rawSnippet,
            dateLastReviewed: row.dateLastReviewed,
          },
        });
        updated++;
      } else {
        const created = await prisma.regulatoryWatch.create({
          data: {
            tenantId,
            theme,
            title: row.title,
            url: row.url,
            modeSuivi: row.modeSuivi,
            typeSource: row.typeSource,
            responsable: row.responsable,
            frequency: row.frequency,
            exploitation: row.exploitation,
            rawSnippet: row.rawSnippet,
            dateLastReviewed: row.dateLastReviewed,
            status: 'ACTIVE',
            suggestedBy: 'IMPORT',
          },
        });
        // AuditLog `regulatoryWatch.created` (actorUserId=null car script CLI)
        await prisma.auditLog.create({
          data: {
            tenantId,
            userId: null,
            entity: 'RegulatoryWatch',
            entityId: created.id,
            action: 'regulatoryWatch.created',
            diff: { source: 'import-xlsx', theme, title: row.title, batch: true } as never,
          },
        });
        inserted++;
      }
    }
  }
  return { inserted, updated, skipped, total };
}

async function main() {
  const xlsxArg = process.argv[2];
  if (!xlsxArg) {
    console.error('Usage: tsx scripts/import-veille-from-xlsx.ts <path-to-xlsx> [tenantId]');
    process.exit(1);
  }
  const tenant = await prisma.tenant.findFirst({ orderBy: { createdAt: 'asc' } });
  if (!tenant) throw new Error('No tenant in DB — run seed first');
  const tenantId = process.argv[3] ?? tenant.id;
  const absPath = path.resolve(xlsxArg);
  console.log(`Importing ${absPath} into tenant ${tenantId}...`);
  const result = await importVeille(tenantId, absPath);
  console.log(`Done: ${result.inserted} inserted, ${result.updated} updated, ${result.skipped} skipped (total ${result.total})`);
}

main().catch((e) => { console.error(e); process.exit(1); });

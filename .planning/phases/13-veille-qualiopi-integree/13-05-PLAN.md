**Étape C — fetch-rss.ts** :
```typescript
import Parser from 'rss-parser';

const parser = new Parser({ timeout: 15_000 });

export interface RssItem {
  title: string;
  link: string;
  pubDate: string | null;
  contentSnippet: string;
}

export async function fetchRssFromSource(url: string): Promise<RssItem[]> {
  try {
    const feed = await parser.parseURL(url);
    return (feed.items ?? []).map((it) => ({
      title: (it.title ?? '').trim(),
      link: (it.link ?? '').trim(),
      pubDate: it.pubDate ?? null,
      contentSnippet: (it.contentSnippet ?? it.content ?? '').trim().slice(0, 800),
    })).filter((it) => it.title && it.link);
  } catch (e) {
    console.warn(`[veille] RSS fetch failed for ${url}:`, (e as Error).message);
    return [];
  }
}
```

**Étape D — classify.ts** (~80 lignes) :

```typescript
import { prisma } from '@qualiof/db';
import { callOllama } from '@/lib/ai-ollama';
import {
  PROMPT_VERSION_VEILLE,
  SYSTEM_PROMPT_VEILLE_CLASSIFY,
  buildVeilleClassifyUserPrompt,
  VeilleClassifyOutputSchema,
} from './prompts';

export interface ClassifyInput { title: string; snippet: string; source: string; }
export interface ClassifyOutput {
  theme: 'INDIC_23' | 'INDIC_24' | 'INDIC_25' | 'INDIC_26' | 'OTHER';
  confidence: number;
  exploitation_draft: string;
}

export async function classifyItem(input: ClassifyInput, tenantId: string): Promise<ClassifyOutput | null> {
  const start = Date.now();
  try {
    const raw = await callOllama({
      model: 'mistral-small:24b',
      systemPrompt: SYSTEM_PROMPT_VEILLE_CLASSIFY,
      prompt: buildVeilleClassifyUserPrompt(input),
      jsonOutput: true,
      temperature: 0.1,
      timeoutMs: 60_000,
    });
    const parsed = VeilleClassifyOutputSchema.safeParse(raw);
    if (!parsed.success) {
      await prisma.aIGenerationJob.create({
        data: {
          tenantId,
          provider: 'ollama',
          model: 'mistral-small:24b',
          promptVersion: PROMPT_VERSION_VEILLE,
          status: 'error',
          latencyMs: Date.now() - start,
          errorMessage: 'JSON parse / Zod fail: ' + JSON.stringify(parsed.error.flatten()).slice(0, 500),
        },
      });
      return null;
    }
    await prisma.aIGenerationJob.create({
      data: {
        tenantId,
        provider: 'ollama',
        model: 'mistral-small:24b',
        promptVersion: PROMPT_VERSION_VEILLE,
        status: parsed.data.theme === 'OTHER' ? 'skipped_other' : 'ok',
        latencyMs: Date.now() - start,
      },
    });
    return parsed.data;
  } catch (e) {
    await prisma.aIGenerationJob.create({
      data: {
        tenantId,
        provider: 'ollama',
        model: 'mistral-small:24b',
        promptVersion: PROMPT_VERSION_VEILLE,
        status: 'error',
        latencyMs: Date.now() - start,
        errorMessage: (e as Error).message?.slice(0, 500),
      },
    });
    return null;
  }
}
```

**Étape E — persist.ts** (~70 lignes) :

```typescript
import { prisma } from '@qualiof/db';
import { logRegulatoryWatchEvent } from '@/lib/regulatoryWatch-audit';
import type { RssItem } from './fetch-rss';
import type { ClassifyOutput } from './classify';

interface PersistInput {
  tenantId: string;
  item: RssItem;
  classification: ClassifyOutput;
  sourceConfig: { name: string; url: string; theme: 'INDIC_23' | 'INDIC_24' | 'INDIC_25' | 'INDIC_26' };
}

export async function persistAutoSuggestion({ tenantId, item, classification, sourceConfig }: PersistInput): Promise<'inserted' | 'skipped_dedup' | 'skipped_other' | 'skipped_low_confidence'> {
  // Guard 1: OTHER → no insert
  if (classification.theme === 'OTHER') return 'skipped_other';

  // Guard 2: confidence < 50 → no insert (paranoid filter; UI inbox shows < 60 with badge "Classification incertaine")
  if (classification.confidence < 50) return 'skipped_low_confidence';

  // Dedup D-11: (tenantId, url, theme) — pas url seule
  const existing = await prisma.regulatoryWatch.findFirst({
    where: { tenantId, url: item.link, theme: classification.theme },
    select: { id: true },
  });
  if (existing) return 'skipped_dedup';

  const created = await prisma.regulatoryWatch.create({
    data: {
      tenantId,
      theme: classification.theme,
      title: item.title.slice(0, 300),
      url: item.link,
      source: sourceConfig.name,
      exploitation: classification.exploitation_draft.slice(0, 5000),
      rawSnippet: item.contentSnippet.slice(0, 2000),
      rssSourceUrl: sourceConfig.url,
      status: 'DRAFT',
      suggestedBy: 'AUTO',
    },
  });

  await logRegulatoryWatchEvent({
    tenantId,
    actorUserId: null, // system
    targetWatchId: created.id,
    action: 'regulatoryWatch.auto_inserted',
    diff: {
      source: sourceConfig.name,
      theme: classification.theme,
      confidence: classification.confidence,
      url: item.link,
    },
  });

  return 'inserted';
}
```

**Étape A — core.ts** (logique pure ingestion) :

```typescript
import { RSS_SOURCES } from './sources';
import { fetchRssFromSource } from './fetch-rss';
import { classifyItem } from './classify';
import { persistAutoSuggestion } from './persist';

export interface IngestResult {
  fetched: number;
  classified: number;
  skipped: number;
  inserted: number;
  errors: number;
}

export async function ingestRssOnceForTenant(tenantId: string): Promise<IngestResult> {
  const result: IngestResult = { fetched: 0, classified: 0, skipped: 0, inserted: 0, errors: 0 };

  for (const source of RSS_SOURCES.filter((s) => s.active)) {
    const items = await fetchRssFromSource(source.url);
    if (items.length === 0) {
      result.errors++;
      continue;
    }
    result.fetched += items.length;

    // Limiter à 5 items les plus récents par source pour ne pas saturer Ollama
    for (const item of items.slice(0, 5)) {
      try {
        const classification = await classifyItem(
          { title: item.title, snippet: item.contentSnippet, source: source.name },
          tenantId,
        );
        if (!classification) {
          result.errors++;
          continue;
        }
        result.classified++;
        const outcome = await persistAutoSuggestion({
          tenantId,
          item,
          classification,
          sourceConfig: { name: source.name, url: source.url, theme: source.theme },
        });
        if (outcome === 'inserted') result.inserted++;
        else result.skipped++;
      } catch (e) {
        console.error('[veille core] item process failed', source.name, item.title, (e as Error).message);
        result.errors++;
      }
    }
  }
  return result;
}
```

**Étape B — queue.ts** :

```typescript
import { Queue } from 'bullmq';
import { getQueueRedis } from '@/lib/closure/redis';

export interface VeilleJobPayload {
  triggered_by: 'cron' | 'manual';
}

let _queue: Queue<VeilleJobPayload> | null = null;

export function getVeilleQueue(): Queue<VeilleJobPayload> {
  if (!_queue) {
    _queue = new Queue<VeilleJobPayload>('veille', { connection: getQueueRedis() });
  }
  return _queue;
}

export async function scheduleWeeklyVeille(): Promise<void> {
  const q = getVeilleQueue();
  await q.add(
    'weekly-veille',
    { triggered_by: 'cron' },
    {
      repeat: { pattern: '0 8 * * 1', tz: 'Europe/Paris' },
      jobId: 'weekly-veille-cron', // idempotence BullMQ
    },
  );
}
```

**Étape C — worker.ts** :

```typescript
import { Worker } from 'bullmq';
import { prisma } from '@qualiof/db';
import { getQueueRedis } from '@/lib/closure/redis';
import { ingestRssOnceForTenant } from './core';

export function startVeilleWorker(): Worker {
  const w = new Worker(
    'veille',
    async (job) => {
      console.log(`[veille worker] job ${job.id} triggered_by=${job.data.triggered_by}`);
      // Multi-tenant : itère sur tous les tenants actifs (ou 1 seul Start Academy en mono-tenant)
      const tenants = await prisma.tenant.findMany({ select: { id: true, name: true } });
      const allResults: any[] = [];
      for (const t of tenants) {
        const r = await ingestRssOnceForTenant(t.id);
        console.log(`[veille worker] tenant ${t.name}: ${JSON.stringify(r)}`);
        allResults.push({ tenantId: t.id, ...r });
      }
      return allResults;
    },
    { connection: getQueueRedis(), concurrency: 1 },
  );
  w.on('failed', (job, err) => console.error('[veille worker] job failed', job?.id, err));
  return w;
}
```

```typescript
import type { User } from '@qualiof/db';

/**
 * D-03 (Phase 13) : LECTEUR n'a PAS accès à l'onglet inbox.
 * Pure helper testable sans rendre la page complète.
 */
export function shouldShowInbox(user: Pick<User, 'role'> | null): boolean {
  if (!user) return false;
  return user.role === 'ADMIN' || user.role === 'MANAGER';
}

export type VeilleTab = 'indic_23' | 'indic_24' | 'indic_25' | 'indic_26' | 'inbox';

export function parseTab(raw: string | undefined, canSeeInbox: boolean): VeilleTab {
  const valid: VeilleTab[] = ['indic_23', 'indic_24', 'indic_25', 'indic_26'];
  if (canSeeInbox) valid.push('inbox');
  return (raw && (valid as string[]).includes(raw)) ? (raw as VeilleTab) : 'indic_23';
}

export function tabToTheme(tab: VeilleTab): 'INDIC_23' | 'INDIC_24' | 'INDIC_25' | 'INDIC_26' | null {
  if (tab === 'inbox') return null;
  return tab.toUpperCase().replace('_', '_') as any;
}
```

```typescript
import { redirect } from 'next/navigation';
import { validateRequest } from '@/lib/auth';
import { prisma } from '@qualiof/db';
import { daysSince } from '@/lib/veille/days-since';
import { shouldShowInbox, parseTab, tabToTheme } from './page-helpers';
import { VeilleTabsClient } from '@/components/veille/veille-tabs-client';
import { VeilleTable } from '@/components/veille/veille-table';
import { VeilleInbox } from '@/components/veille/veille-inbox';
import { DaysSinceBadge } from '@/components/veille/days-since-badge';
import { AddVeilleDialog } from '@/components/veille/add-veille-dialog';
import { ExportPdfButton } from '@/components/veille/export-pdf-button'; // créé Plan 04

export const dynamic = 'force-dynamic';

interface SearchParams {
  tab?: string;
  q?: string;
  responsable?: string;
  freq?: string;
}

export default async function VeillePage({ searchParams }: { searchParams: Promise<SearchParams> }) {
  const { user } = await validateRequest();
  if (!user) redirect('/login');

  const sp = await searchParams;
  const canSeeInbox = shouldShowInbox(user);
  const tab = parseTab(sp.tab, canSeeInbox);
  const canEdit = user.role === 'ADMIN' || user.role === 'MANAGER';

  // Inbox count (sidebar badge) — seulement si canSeeInbox
  const inboxCount = canSeeInbox
    ? await prisma.regulatoryWatch.count({
        where: { tenantId: user.tenantId, status: 'DRAFT', suggestedBy: 'AUTO' },
      })
    : 0;

  if (tab === 'inbox') {
    if (!canSeeInbox) redirect('/app/veille?tab=indic_23'); // defense-in-depth
    const suggestions = await prisma.regulatoryWatch.findMany({
      where: { tenantId: user.tenantId, status: 'DRAFT', suggestedBy: 'AUTO' },
      orderBy: { createdAt: 'desc' },
    });
    return (
      <main className="p-4 md:p-8">
        <Header user={user} canEdit={canEdit} />
        <VeilleTabsClient activeTab={tab} canSeeInbox={canSeeInbox} inboxCount={inboxCount} />
        <VeilleInbox suggestions={suggestions} />
      </main>
    );
  }

  const theme = tabToTheme(tab)!; // never null for non-inbox tab
  const filters: any = {
    tenantId: user.tenantId,
    theme,
    status: canEdit ? { in: ['ACTIVE', 'DRAFT'] } : 'ACTIVE', // LECTEUR ne voit que ACTIVE
  };
  if (sp.responsable) filters.responsable = { contains: sp.responsable, mode: 'insensitive' };
  if (sp.freq) filters.frequency = { contains: sp.freq, mode: 'insensitive' };
  if (sp.q) {
    filters.OR = [
      { title: { contains: sp.q, mode: 'insensitive' } },
      { source: { contains: sp.q, mode: 'insensitive' } },
    ];
  }

  const [watches, lastReview] = await Promise.all([
    prisma.regulatoryWatch.findMany({
      where: filters,
      orderBy: [{ dateLastReviewed: 'desc' }, { dateAdded: 'desc' }],
    }),
    prisma.regulatoryWatch.findFirst({
      where: { tenantId: user.tenantId, theme, status: 'ACTIVE' },
      orderBy: { dateLastReviewed: 'desc' },
      select: { dateLastReviewed: true },
    }),
  ]);

  const days = daysSince(lastReview?.dateLastReviewed ?? null);

  return (
    <main className="p-4 md:p-8 space-y-4">
      <Header user={user} canEdit={canEdit} />
      <VeilleTabsClient activeTab={tab} canSeeInbox={canSeeInbox} inboxCount={inboxCount} />
      <div className="flex items-center justify-between flex-wrap gap-2">
        <DaysSinceBadge days={days} />
        {canEdit && <ExportPdfButton theme={theme} />}
      </div>
      <VeilleTable watches={watches} canEdit={canEdit} theme={theme} />
    </main>
  );
}

function Header({ user, canEdit }: { user: any; canEdit: boolean }) {
  return (
    <header className="flex items-start justify-between gap-4 flex-wrap">
      <div>
        <h1 className="text-2xl font-bold">Veille Qualiopi</h1>
        <p className="text-sm text-muted-foreground">Critère 6 — indicateurs 23 / 24 / 25 / 26</p>
      </div>
      {canEdit && <AddVeilleDialog />}
    </header>
  );
}
```

**Étape A — VeilleTabsClient** (`apps/web/src/components/veille/veille-tabs-client.tsx`)

Client component qui rend les onglets et synchronise avec l'URL :

```tsx
'use client';
import { usePathname, useRouter, useSearchParams } from 'next/navigation';
import { cn } from '@/lib/utils';
import { Inbox } from 'lucide-react';

const TABS = [
  { id: 'indic_23', label: '23 — Formation pro' },
  { id: 'indic_24', label: '24 — Secteur d\'activité' },
  { id: 'indic_25', label: '25 — Innovations péda' },
  { id: 'indic_26', label: '26 — Handicap & DREETS' },
] as const;

interface Props {
  activeTab: string;
  canSeeInbox: boolean;
  inboxCount: number;
}

export function VeilleTabsClient({ activeTab, canSeeInbox, inboxCount }: Props) {
  const router = useRouter();
  const pathname = usePathname();
  const sp = useSearchParams();

  const setTab = (id: string) => {
    const params = new URLSearchParams(sp.toString());
    params.set('tab', id);
    // Reset filters when changing thématique tab (preserve only ?q if any)
    if (id !== activeTab) {
      params.delete('responsable');
      params.delete('freq');
    }
    router.push(`${pathname}?${params.toString()}`);
  };

  return (
    <nav className="flex items-center gap-1 border-b">
      {TABS.map((t) => (
        <button
          key={t.id}
          onClick={() => setTab(t.id)}
          className={cn(
            'px-3 py-2 text-sm font-medium border-b-2 transition-colors',
            activeTab === t.id ? 'border-primary text-primary' : 'border-transparent text-muted-foreground hover:text-foreground',
          )}
        >
          {t.label}
        </button>
      ))}
      {canSeeInbox && (
        // D-03 : Strictly NOT rendered for LECTEUR — server already filters via shouldShowInbox prop
        <button
          onClick={() => setTab('inbox')}
          className={cn(
            'px-3 py-2 text-sm font-medium border-b-2 transition-colors ml-auto flex items-center gap-1.5',
            activeTab === 'inbox' ? 'border-amber-500 text-amber-700' : 'border-transparent text-muted-foreground hover:text-foreground',
          )}
        >
          <Inbox className="size-4" aria-hidden />
          Inbox
          {inboxCount > 0 && (
            <span className="text-xs bg-amber-100 text-amber-800 rounded-full px-1.5 py-0.5">{inboxCount}</span>
          )}
        </button>
      )}
    </nav>
  );
```

```tsx
'use client';
import { useState, useTransition } from 'react';
import { Button } from '@/components/ui/button';
import { updateExploitation } from '@/server/actions/veille';
import { toast } from 'sonner';
import { Pencil, Save, X } from 'lucide-react';

export function ExploitationCell({ id, value }: { id: string; value: string | null }) {
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState(value ?? '');
  const [pending, startTransition] = useTransition();

  if (!editing) {
    return (
      <div className="flex items-start gap-2">
        <p className="text-sm flex-1 line-clamp-2" title={value ?? ''}>{value ?? <span className="text-muted-foreground italic">À renseigner</span>}</p>
        <button
          onClick={() => setEditing(true)}
          className="opacity-0 group-hover:opacity-100 hover:opacity-100 text-muted-foreground hover:text-foreground"
          aria-label="Éditer l'exploitation"
        >
          <Pencil className="size-4" />
        </button>
      </div>
    );
  }

  const onSave = () => {
    startTransition(async () => {
      const r = await updateExploitation({ id, exploitation: draft.trim() });
      if (r.ok) {
        toast.success('Exploitation mise à jour');
        setEditing(false);
      } else {
        toast.error(r.error || 'Erreur');
      }
    });
  };

  return (
    <div className="space-y-2">
      <textarea
        value={draft}
        onChange={(e) => setDraft(e.target.value)}
        className="w-full min-h-[80px] text-sm border rounded p-2"
        autoFocus
      />
      <div className="flex gap-2">
        <Button size="sm" onClick={onSave} disabled={pending || draft.trim().length === 0}>
          <Save className="size-3 mr-1" /> Enregistrer
        </Button>
        <Button size="sm" variant="ghost" onClick={() => { setDraft(value ?? ''); setEditing(false); }} disabled={pending}>
          <X className="size-3 mr-1" /> Annuler
        </Button>
      </div>
    </div>
  );
}
```

**Étape G — VeilleInbox** (`apps/web/src/components/veille/veille-inbox.tsx`)

Liste de cards (1 par DRAFT/AUTO suggestion) avec : titre + url + theme proposé (badge) + exploitation_draft (rawSnippet ou exploitation) + 2 boutons :
- "Valider" → appelle `approveWatch(id)`
- "Rejeter" → ouvre AlertDialog avec textarea reason → appelle `rejectWatch({ id, reason })`

```tsx
'use client';
import { useState, useTransition } from 'react';
import type { RegulatoryWatch } from '@qualiof/db';
import { approveWatch, rejectWatch } from '@/server/actions/veille';
import { toast } from 'sonner';
import { Button } from '@/components/ui/button';
import { Check, X } from 'lucide-react';

export function VeilleInbox({ suggestions }: { suggestions: RegulatoryWatch[] }) {
  if (suggestions.length === 0) {
    return <p className="text-sm text-muted-foreground py-8">Aucune suggestion auto en attente.</p>;
  }
  return (
    <div className="space-y-3">
      {suggestions.map((s) => <InboxCard key={s.id} watch={s} />)}
    </div>
  );
}

function InboxCard({ watch }: { watch: RegulatoryWatch }) {
  const [pending, startTransition] = useTransition();
  const [rejectOpen, setRejectOpen] = useState(false);
  const [reason, setReason] = useState('');

  const onApprove = () => startTransition(async () => {
    const r = await approveWatch(watch.id);
    if (r.ok) toast.success('Suggestion validée');
    else toast.error(r.error || 'Erreur');
  });

  const onReject = () => startTransition(async () => {
    if (reason.trim().length < 2) { toast.error('Raison requise (≥ 2 caractères)'); return; }
    const r = await rejectWatch({ id: watch.id, reason: reason.trim() });
    if (r.ok) { toast.success('Suggestion rejetée'); setRejectOpen(false); setReason(''); }
    else toast.error(r.error || 'Erreur');
  });

  return (
    <div className="border rounded-lg p-4 space-y-2 bg-amber-50/50">
      <header className="flex items-start justify-between gap-2">
        <div>
          <span className="text-xs font-medium px-2 py-0.5 rounded bg-amber-100 text-amber-800">{watch.theme}</span>
          <h3 className="font-medium mt-1">{watch.title}</h3>
          {watch.url && <a href={watch.url} target="_blank" rel="noopener" className="text-xs text-blue-600 hover:underline">{watch.url}</a>}
        </div>
        <div className="flex gap-2">
          <Button size="sm" onClick={onApprove} disabled={pending}><Check className="size-3 mr-1" /> Valider</Button>
          <Button size="sm" variant="outline" onClick={() => setRejectOpen(!rejectOpen)} disabled={pending}><X className="size-3 mr-1" /> Rejeter</Button>
        </div>
      </header>
      {watch.exploitation && <p className="text-sm text-muted-foreground">{watch.exploitation}</p>}
      {watch.rawSnippet && <p className="text-xs text-muted-foreground italic">{watch.rawSnippet}</p>}
      {rejectOpen && (
        <div className="pt-2 border-t space-y-2">
          <textarea
            value={reason}
            onChange={(e) => setReason(e.target.value)}
            placeholder="Raison du rejet (visible dans l'audit log)"
            className="w-full text-sm border rounded p-2 min-h-[60px]"
          />
          <Button size="sm" variant="destructive" onClick={onReject} disabled={pending || reason.trim().length < 2}>Confirmer le rejet</Button>
        </div>
      )}
    </div>
  );
}
```

```typescript
'use server';

import { prisma } from '@qualiof/db';
import {
  createWatchSchema,
  updateWatchSchema,
  updateExploitationSchema,
  rejectWatchSchema,
  type CreateWatchInput,
  type UpdateWatchInput,
} from '@qualiof/shared';
import { requireRole, ForbiddenError } from '@/lib/rbac';
import { logRegulatoryWatchEvent } from '@/lib/regulatoryWatch-audit';
import { revalidatePath } from 'next/cache';

const ROUTE = '/app/veille';

type ActionResult<T = void> = (T extends void ? { ok: true } : { ok: true } & T) | { ok: false; error: string };

export async function createWatch(input: CreateWatchInput): Promise<ActionResult<{ watchId: string }>> {
  try {
    const user = await requireRole(['ADMIN', 'MANAGER']);
    const parsed = createWatchSchema.safeParse(input);
    if (!parsed.success) return { ok: false, error: parsed.error.flatten().formErrors.join(', ') || 'Validation failed' };
    const created = await prisma.regulatoryWatch.create({
      data: {
        tenantId: user.tenantId,
        ...parsed.data,
        status: 'ACTIVE',
        suggestedBy: 'USER',
      },
    });
    await logRegulatoryWatchEvent({
      tenantId: user.tenantId,
      actorUserId: user.id,
      targetWatchId: created.id,
      action: 'regulatoryWatch.created',
      diff: { source: 'manual-ui', theme: parsed.data.theme, title: parsed.data.title },
    });
    revalidatePath(ROUTE);
    return { ok: true, watchId: created.id };
  } catch (e) {
    if (e instanceof ForbiddenError) return { ok: false, error: 'Forbidden — ADMIN ou MANAGER requis' };
    console.error('[createWatch] error', e);
    return { ok: false, error: (e as Error).message ?? 'unknown error' };
  }
}

export async function updateWatch(input: UpdateWatchInput): Promise<ActionResult> {
  try {
    const user = await requireRole(['ADMIN', 'MANAGER']);
    const parsed = updateWatchSchema.safeParse(input);
    if (!parsed.success) return { ok: false, error: 'Validation failed' };
    const { id, ...data } = parsed.data;
    const existing = await prisma.regulatoryWatch.findFirst({ where: { id, tenantId: user.tenantId } });
    if (!existing) return { ok: false, error: 'Not found or wrong tenant' };
    // Compute diff (excluding exploitation — that's updateExploitation's job)
    const diff: Record<string, { before: unknown; after: unknown }> = {};
    for (const [k, v] of Object.entries(data)) {
      if (k === 'exploitation') continue;
      if (v !== undefined && (existing as any)[k] !== v) {
        diff[k] = { before: (existing as any)[k], after: v };
      }
    }
    await prisma.regulatoryWatch.update({ where: { id }, data });
    if (Object.keys(diff).length > 0) {
      await logRegulatoryWatchEvent({
        tenantId: user.tenantId, actorUserId: user.id, targetWatchId: id,
        action: 'regulatoryWatch.updated', diff,
      });
    }
    revalidatePath(ROUTE);
    return { ok: true };
  } catch (e) {
    if (e instanceof ForbiddenError) return { ok: false, error: 'Forbidden' };
    return { ok: false, error: (e as Error).message };
  }
}

export async function updateExploitation(input: { id: string; exploitation: string }): Promise<ActionResult> {
  try {
    const user = await requireRole(['ADMIN', 'MANAGER']);
    const parsed = updateExploitationSchema.safeParse(input);
    if (!parsed.success) return { ok: false, error: 'Validation failed' };
    const existing = await prisma.regulatoryWatch.findFirst({
      where: { id: parsed.data.id, tenantId: user.tenantId },
      select: { id: true, exploitation: true },
    });
    if (!existing) return { ok: false, error: 'Not found' };
    const now = new Date();
    await prisma.regulatoryWatch.update({
      where: { id: parsed.data.id },
      data: { exploitation: parsed.data.exploitation, dateLastReviewed: now },
    });
    await logRegulatoryWatchEvent({
      tenantId: user.tenantId, actorUserId: user.id, targetWatchId: parsed.data.id,
      action: 'regulatoryWatch.exploitation_updated',
      diff: {
        exploitation: { before: existing.exploitation, after: parsed.data.exploitation },
        dateLastReviewed: { before: null, after: now.toISOString() },
      },
    });
    revalidatePath(ROUTE);
    return { ok: true };
  } catch (e) {
    if (e instanceof ForbiddenError) return { ok: false, error: 'Forbidden' };
    return { ok: false, error: (e as Error).message };
  }
}

export async function approveWatch(watchId: string): Promise<ActionResult> {
  try {
    const user = await requireRole(['ADMIN', 'MANAGER']);
    const existing = await prisma.regulatoryWatch.findFirst({
      where: { id: watchId, tenantId: user.tenantId },
      select: { id: true, status: true, suggestedBy: true },
    });
    if (!existing) return { ok: false, error: 'Not found' };
    if (existing.suggestedBy !== 'AUTO' || existing.status !== 'DRAFT') {
      return { ok: false, error: 'Only DRAFT/AUTO suggestions can be approved' };
    }
    await prisma.regulatoryWatch.update({
      where: { id: watchId },
      data: { status: 'ACTIVE' },
    });
    await logRegulatoryWatchEvent({
      tenantId: user.tenantId, actorUserId: user.id, targetWatchId: watchId,
      action: 'regulatoryWatch.approved',
      diff: { status: { before: 'DRAFT', after: 'ACTIVE' } },
    });
    revalidatePath(ROUTE);
    return { ok: true };
  } catch (e) {
    if (e instanceof ForbiddenError) return { ok: false, error: 'Forbidden' };
    return { ok: false, error: (e as Error).message };
  }
}

export async function rejectWatch(input: { id: string; reason: string }): Promise<ActionResult> {
  try {
    const user = await requireRole(['ADMIN', 'MANAGER']);
    const parsed = rejectWatchSchema.safeParse(input);
    if (!parsed.success) return { ok: false, error: 'Validation failed' };
    const existing = await prisma.regulatoryWatch.findFirst({
      where: { id: parsed.data.id, tenantId: user.tenantId },
      select: { id: true, status: true, suggestedBy: true },
    });
    if (!existing) return { ok: false, error: 'Not found' };
    if (existing.suggestedBy !== 'AUTO') {
      return { ok: false, error: 'Only AUTO suggestions can be rejected via inbox' };
    }
    await prisma.regulatoryWatch.update({
      where: { id: parsed.data.id },
      data: { status: 'ARCHIVED' },
    });
    await logRegulatoryWatchEvent({
      tenantId: user.tenantId, actorUserId: user.id, targetWatchId: parsed.data.id,
      action: 'regulatoryWatch.rejected',
      diff: {
        status: { before: existing.status, after: 'ARCHIVED' },
        rejectionReason: parsed.data.reason,
      },
    });
    revalidatePath(ROUTE);
    return { ok: true };
  } catch (e) {
    if (e instanceof ForbiddenError) return { ok: false, error: 'Forbidden' };
    return { ok: false, error: (e as Error).message };
  }
}

export async function archiveWatch(watchId: string): Promise<ActionResult> {
  try {
    const user = await requireRole(['ADMIN', 'MANAGER']);
    const existing = await prisma.regulatoryWatch.findFirst({
      where: { id: watchId, tenantId: user.tenantId },
      select: { id: true, status: true },
    });
    if (!existing) return { ok: false, error: 'Not found' };
    await prisma.regulatoryWatch.update({ where: { id: watchId }, data: { status: 'ARCHIVED' } });
    await logRegulatoryWatchEvent({
      tenantId: user.tenantId, actorUserId: user.id, targetWatchId: watchId,
      action: 'regulatoryWatch.archived',
      diff: { status: { before: existing.status, after: 'ARCHIVED' } },
    });
    revalidatePath(ROUTE);
    return { ok: true };
```

      { type: 'PROGRAMME', indicator: 'Indicateur 9' },
      { type: 'DEROULE_PEDAGOGIQUE', indicator: 'Indicateur 9' },

    ];

    describe('QualiopiDocCatalog seed — Ind 9/11/12/30 mapping (Phase 10 D-08)', () => {
      it.each(EXPECTED_MAPPING)(
        '$type → $indicator',
        async ({ type, indicator }) => {
          const entry = await prisma.qualiopiDocCatalog.findUnique({
            where: { type: type as never },
          });
          expect(entry).not.toBeNull();
          expect(entry!.qualiopiIndicator).toBe(indicator);
        },
      );

      it("aucun des 8 DocType auto-calculables n'a qualiopiIndicator NULL", async () => {
        const types = EXPECTED_MAPPING.map((e) => e.type);
        const entries = await prisma.qualiopiDocCatalog.findMany({
          where: { type: { in: types as never } },
        });
        expect(entries).toHaveLength(types.length);
        for (const e of entries) {
          expect(e.qualiopiIndicator).not.toBeNull();
        }
      });
    });

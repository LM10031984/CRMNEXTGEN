    export function getById(number: number): QualiopiIndicatorEntry | undefined {
      return QUALIOPI_INDICATORS_CATALOG.find((e) => e.number === number);
    }

    export function getByCriterion(criterion: number): QualiopiIndicatorEntry[] {
      return QUALIOPI_INDICATORS_CATALOG.filter((e) => e.criterion === criterion);
    }

    export function getBySource(source: IndicatorSource): QualiopiIndicatorEntry[] {
      return QUALIOPI_INDICATORS_CATALOG.filter((e) => e.source === source);
    }

    export function countByStatus(
      entries: Array<{ indicatorNumber: number; status: string }>,
    ): Record<string, number> {
      const counts: Record<string, number> = {
        CONFORME: 0,
        NON_CONFORME: 0,
        EN_COURS: 0,
        NA: 0,
      };
      for (const e of entries) {
        if (counts[e.status] !== undefined) counts[e.status]++;
      }
      return counts;
    }

    /** Numéro vers libellé court — utile pour PDF et UI. */
    export function getLabel(number: number): string {
      return getById(number)?.label ?? `Indicateur ${number}`;
    }

      | 'qualiopiBlanc.auto_recalculated'
      | 'qualiopiBlanc.exported'
      | 'qualiopiBlanc.notification_sent';

    export async function logQualiopiBlancEvent(opts: {
      tenantId: string;
      actorUserId: string | null;
      targetEntryId: string; // QualiopiBlancEntry.id OU 'BULK' OU 'SESSION:{sessionId}'
      action: QualiopiBlancAction;
      diff?: Diff | Record<string, unknown>;

        data: {
          tenantId: opts.tenantId,
          userId: opts.actorUserId,
          entity: 'QualiopiBlancEntry',
          entityId: opts.targetEntryId,
          action: opts.action,
          diff: (opts.diff ?? {}) as never,
        },
      });
    }

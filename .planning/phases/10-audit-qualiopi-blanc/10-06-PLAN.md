    ```typescript
    /**
     * Helper pur (worker-safe) : calcule la complétude de la matrice docs
     * Qualiopi d'une session.
     *
     * Pour chaque participant × chaque DocType MANDATORY, compte si présent
     * (via Document.entityType='participant' OU dérivé docStatus).
     *
     * Pour chaque DocType session-only (DEROULE_PEDAGOGIQUE, CHECKLIST_FORMATION,
     * GRILLE_OBS_SESSION), compte si présent (Document.entityType='session').
     *
     * Utilisé par :
     *  - Worker daily Plan 10-07 (détecter sessions endDate ∈ [now+6j, now+7j]
     *    avec percent < 100)
     *  - UI Plan 10-06 (afficher % par session dans la liste)
     *
     * Worker safety : 0 import server-actions/rbac/React/next.
     */

    import type { DocType } from '@qualiof/db';

    /** DocType matrice participant (14 docs — clone de MATRIX_DOC_TYPES Phase 9.1).
     *  Hard-coded ici pour éviter le coupling Phase 9.1 → si Phase 9.1 ajoute des
     *  DocTypes, regarder si Phase 10 doit aussi évoluer. */
    export const PARTICIPANT_MANDATORY_DOCS: readonly DocType[] = [
      'CONVENTION',
      'PROGRAMME',
      'CONVOCATION',
      'EMARGEMENT',
      'ASSIDUITE',
      'EVALUATION_ACQUIS',
      'ATTESTATION_FIN',
      'CERTIFICAT_REALISATION',
      'SATISFACTION',
    ];

    export const SESSION_LEVEL_DOCS: readonly DocType[] = [
      'DEROULE_PEDAGOGIQUE',
      'CHECKLIST_FORMATION',
      'GRILLE_OBS_SESSION',
    ];

    export interface SessionCompletenessInput {
      sessionId: string;
      sessionLabel: string;
      endDate: Date | null;
      participantIds: string[];
      /** Map participantId → DocType[] présents (Document.entityType='participant'). */
      participantDocsPresent: Map<string, Set<DocType>>;
      /** DocType[] présents au niveau session. */
      sessionDocsPresent: Set<DocType>;
    }

    export interface SessionCompletenessResult {
      sessionId: string;
      sessionLabel: string;
      endDate: Date | null;
      participantCount: number;
      /** Nombre total de cellules attendues (participants × PARTICIPANT_MANDATORY_DOCS + SESSION_LEVEL_DOCS). */
      expectedCells: number;
      /** Cellules présentes. */
      filledCells: number;
      /** Cellules manquantes (expectedCells - filledCells). */
      missingCount: number;
      /** Pourcentage [0..100]. */
      percent: number;
      /** Liste des DocType manquants au moins une fois (pour email récap). */
      missingDocTypes: DocType[];
    }

    export function computeSessionCompleteness(
      input: SessionCompletenessInput,
    ): SessionCompletenessResult {
      const participantCount = input.participantIds.length;

      // Cells attendues
      const expectedCells = participantCount * PARTICIPANT_MANDATORY_DOCS.length + SESSION_LEVEL_DOCS.length;

      // Cells participantes remplies
      let filledCells = 0;
      const missingTypeSet = new Set<DocType>();
      for (const pId of input.participantIds) {
        const docs = input.participantDocsPresent.get(pId) ?? new Set<DocType>();
        for (const dt of PARTICIPANT_MANDATORY_DOCS) {
          if (docs.has(dt)) filledCells++;
          else missingTypeSet.add(dt);
        }
      }

      // Cells session-only remplies
      for (const dt of SESSION_LEVEL_DOCS) {
        if (input.sessionDocsPresent.has(dt)) filledCells++;
        else missingTypeSet.add(dt);
      }

      const missingCount = expectedCells - filledCells;
      const percent = expectedCells === 0 ? 0 : Math.round((filledCells / expectedCells) * 100);

      return {
        sessionId: input.sessionId,
        sessionLabel: input.sessionLabel,
        endDate: input.endDate,
        participantCount,
        expectedCells,
        filledCells,
        missingCount,
        percent,
        missingDocTypes: Array.from(missingTypeSet),
      };
    }
    ```

    ```tsx
    import { redirect } from 'next/navigation';
    import { Breadcrumb } from '@/components/ui/breadcrumb';
    import { requireRole, UnauthorizedError, ForbiddenError } from '@/lib/rbac';
    import { prisma } from '@qualiof/db';
    import {
      computeSessionCompleteness,
      PARTICIPANT_MANDATORY_DOCS,
      SESSION_LEVEL_DOCS,
    } from '@/lib/qualiopi-blanc/session-completeness';
    import { SessionsList } from '@/components/audit-qualiopi-blanc/sessions-list';

    export const dynamic = 'force-dynamic';

    export default async function AuditSessionsListPage() {
      let user;
      try {
        user = await requireRole(['ADMIN', 'MANAGER', 'LECTEUR']);
      } catch (e) {
        if (e instanceof UnauthorizedError) redirect('/login');
        if (e instanceof ForbiddenError) redirect('/app');
        throw e;
      }

      const sessions = await prisma.trainingSession.findMany({
        where: { tenantId: user.tenantId },
        select: {
          id: true,
          title: true,
          startDate: true,
          endDate: true,
          participants: { select: { id: true } },
          documents: {
            select: { type: true, participantId: true, entityType: true },
          },
        },
        orderBy: { endDate: 'desc' },
        take: 200, // Pagination V2
      });

      const results = sessions.map((s) => {
        const participantDocsPresent = new Map<string, Set<any>>();
        const sessionDocsPresent = new Set<any>();
        for (const p of s.participants) participantDocsPresent.set(p.id, new Set());
        for (const d of s.documents) {
          if (d.entityType === 'session' && !d.participantId) {
            sessionDocsPresent.add(d.type);
          }
          if (d.entityType === 'participant' && d.participantId) {
            const set = participantDocsPresent.get(d.participantId);
            if (set) set.add(d.type);
          }
        }
        return computeSessionCompleteness({
          sessionId: s.id,
          sessionLabel: s.title,
          endDate: s.endDate,
          participantIds: s.participants.map((p) => p.id),
          participantDocsPresent,
          sessionDocsPresent,
        });
      });

      return (
        <div className="space-y-6 p-4 md:p-8">
          <Breadcrumb
            items={[
              { label: 'Tableau de bord', href: '/app' },
              { label: 'Audit Qualiopi blanc', href: '/app/audit-qualiopi-blanc' },
              { label: 'Sessions' },
            ]}
          />
          <div>
            <h1 className="text-2xl font-semibold tracking-tight">Sessions — complétude matrice docs</h1>
            <p className="text-sm text-muted-foreground">
              Pour chaque session : % de la matrice docs Qualiopi rempli (
              {PARTICIPANT_MANDATORY_DOCS.length} docs par participant + {SESSION_LEVEL_DOCS.length} docs
              session). Cliquer pour drill-down.
            </p>
          </div>
          <SessionsList results={results} />
        </div>
      );
    }
    ```

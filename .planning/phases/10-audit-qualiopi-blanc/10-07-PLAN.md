    import { Queue } from 'bullmq';
    import { getQueueRedis } from '../closure/redis';

    export const QUALIOPI_BLANC_QUEUE_NAME = 'qualiopi-blanc-alerts';

    export interface QualiopiBlancJobPayload {
      triggered_by: 'cron' | 'manual';
    }

    let _queue: Queue<QualiopiBlancJobPayload> | null = null;

    export function getQualiopiBlancQueue(): Queue<QualiopiBlancJobPayload> {
      if (_queue) return _queue;
      _queue = new Queue<QualiopiBlancJobPayload>(QUALIOPI_BLANC_QUEUE_NAME, {
        connection: getQueueRedis(),
        defaultJobOptions: {
          attempts: 2,
          backoff: { type: 'exponential', delay: 60_000 },
          removeOnComplete: { count: 30 },
          removeOnFail: { count: 50 },
        },
      });
      return _queue;
    }

    /**
     * Inscrit le job repeatable cron daily (9h Europe/Paris). Idempotent grâce
     * au `jobId` fixe (BullMQ dédoublonne en interne).
     *
     * Doit être appelé au boot du process worker (cf `scripts/qualiopi-blanc-worker.ts`).
     */
    export async function scheduleDailyQualiopiBlanc(): Promise<void> {
      const queue = getQualiopiBlancQueue();
      await queue.add(
        'daily-qualiopi-blanc',
        { triggered_by: 'cron' as const },
        {
          repeat: { pattern: '0 9 * * *', tz: 'Europe/Paris' },
          jobId: 'daily-qualiopi-blanc-alerts',
        },
      );
      console.log(
        '[qualiopi-blanc] daily cron registered (09:00 Europe/Paris, jobId=daily-qualiopi-blanc-alerts)',
      );
    }

    /**
     * Phase 10 — BullMQ Worker Audit Qualiopi blanc (QBLANC-02 — alertes 7j).
     *
     * Pour chaque tenant :
     *  1. recalcAutoAllForTenant (Plan 10-03) → MAJ statuts auto Ind 9/11/12/30
     *  2. Liste les sessions endDate ∈ [now+6j, now+7j] (24h window pour gérer
     *     l'idempotence d'une exécution daily)
     *  3. Pour chaque session : computeSessionCompleteness
     *  4. Filtre les sessions incomplètes (percent < 100)
     *  5. Si >= 1 session : pour chaque user ADMIN+MANAGER, crée Notification cloche
     *     + envoie 1 email récap groupé (1 email par user, listant N sessions)
     *  6. AuditLog `qualiopiBlanc.notification_sent` (targetEntryId='SESSION:{sessionId}')
     *
     * Idempotence : la window 24h fait que une session sera détectée 1 seule fois.
     * Si re-run le même jour : Notification + email seront recrées (acceptable
     * pour V1 — V2 = dedupe par sessionId + date).
     *
     * Multi-tenant : itère sur tous les tenants (Start Academy mono-tenant en V1).
     *
     * Worker safety NON-NÉGOCIABLE : 0 import server-actions/rbac/React/next.
     */

    import { Worker, type Job } from 'bullmq';
    import { prisma } from '@qualiof/db';
    import { getWorkerRedis } from '../closure/redis';
    import {
      QUALIOPI_BLANC_QUEUE_NAME,
      type QualiopiBlancJobPayload,
    } from './queue';
    import { recalcAutoAllForTenant } from './core';
    import {
      computeSessionCompleteness,
      type SessionCompletenessResult,
    } from './session-completeness';
    import { renderSessionIncompleteEmail } from './notification-email';
    import { logQualiopiBlancEvent } from '../qualiopi-blanc-audit';
    import { sendMail } from '../mailer';

    interface TenantTickSummary {
      tenantId: string;
      tenantName: string;
      recalcUpdated: number;
      sessionsDetected: number;
      notificationsCreated: number;
      emailsSent: number;
      emailsDryRun: number;
    }

    /**
     * Liste les sessions d'un tenant dont endDate ∈ [now+6j, now+7j].
     * Window 24h pour idempotence quotidienne.
     */
    async function getSessionsClosingIn7Days(tenantId: string): Promise<SessionCompletenessResult[]> {
      const now = new Date();
      const sixDaysFromNow = new Date(now.getTime() + 6 * 24 * 60 * 60 * 1000);
      const sevenDaysFromNow = new Date(now.getTime() + 7 * 24 * 60 * 60 * 1000);

      const sessions = await prisma.trainingSession.findMany({
        where: {
          tenantId,
          endDate: { gte: sixDaysFromNow, lte: sevenDaysFromNow },
        },
        select: {
          id: true,
          title: true,
          endDate: true,
          participants: { select: { id: true } },
          documents: {
            select: { type: true, participantId: true, entityType: true },
          },
        },
      });

      const results: SessionCompletenessResult[] = sessions.map((s) => {
        const participantDocsPresent = new Map<string, Set<any>>();
        const sessionDocsPresent = new Set<any>();
        for (const p of s.participants) participantDocsPresent.set(p.id, new Set());
        for (const d of s.documents) {
          if (d.entityType === 'session' && !d.participantId) sessionDocsPresent.add(d.type);
          if (d.entityType === 'participant' && d.participantId) {
            const set = participantDocsPresent.get(d.participantId);
            if (set) set.add(d.type);
          }
        }
        return computeSessionCompleteness({
          sessionId: s.id,
          sessionLabel: s.title,
          endDate: s.endDate,
          participantIds: s.participants.map((p) => p.id),
          participantDocsPresent,
          sessionDocsPresent,
        });
      });

      // Garde seulement les sessions incomplètes
      return results.filter((r) => r.percent < 100);
    }

    export async function processQualiopiBlancJob(
      job: Job<QualiopiBlancJobPayload>,
    ): Promise<TenantTickSummary[]> {
      console.log(`[qualiopi-blanc-worker] tick jobId=${job.id} triggered_by=${job.data.triggered_by}`);

      const tenants = await prisma.tenant.findMany({ select: { id: true, name: true } });
      const summaries: TenantTickSummary[] = [];
      const appBaseUrl = process.env.APP_BASE_URL ?? 'http://localhost:3010';

      for (const tenant of tenants) {
        // 1. Recalcule les indicateurs auto
        const recalc = await recalcAutoAllForTenant(tenant.id, null);

        // 2 + 3 + 4. Détecte sessions incomplètes à J+7
        const incompleteSessions = await getSessionsClosingIn7Days(tenant.id);

        let notificationsCreated = 0;
        let emailsSent = 0;
        let emailsDryRun = 0;

        if (incompleteSessions.length > 0) {
          // Recipients : tous les users ADMIN + MANAGER actifs du tenant
          const recipients = await prisma.user.findMany({
            where: {
              tenantId: tenant.id,
              role: { in: ['ADMIN', 'MANAGER'] },
              disabledAt: null,
            },
            select: { id: true, email: true, firstName: true, lastName: true },
          });

          for (const user of recipients) {
            // 5a. Crée 1 Notification cloche par session (pour permettre drill-down)
            for (const sess of incompleteSessions) {
              await prisma.notification.create({
                data: {
                  tenantId: tenant.id,
                  userId: user.id,
                  type: 'qualiopiBlanc.sessionIncomplete',
                  payload: {
                    sessionId: sess.sessionId,
                    sessionLabel: sess.sessionLabel,
                    endDate: sess.endDate?.toISOString() ?? null,
                    missingDocsCount: sess.missingCount,
                    missingIndicatorsCount: 0, // V1 — V2 calcul depuis catalogue
                    drillDownUrl: `/app/audit-qualiopi-blanc/sessions/${sess.sessionId}`,
                  } as never,
                },
              });
              notificationsCreated++;
            }

            // 5b. Envoie 1 email récap groupé (1 email par user, listant les N sessions)
            const { subject, html, text } = renderSessionIncompleteEmail({
              recipientName: `${user.firstName} ${user.lastName}`.trim(),
              tenantName: tenant.name,
              sessions: incompleteSessions,
              appBaseUrl,
            });
            const mailResult = await sendMail({
              to: user.email,
              subject,
              html,
              text,
            });
            if (mailResult.ok && mailResult.dryRun) emailsDryRun++;
            else if (mailResult.ok) emailsSent++;
          }

          // 6. AuditLog 1 ligne par session détectée
          for (const sess of incompleteSessions) {
            await logQualiopiBlancEvent({
              tenantId: tenant.id,
              actorUserId: null,
              targetEntryId: `SESSION:${sess.sessionId}`,
              action: 'qualiopiBlanc.notification_sent',
              diff: {
                sessionId: sess.sessionId,
                sessionLabel: sess.sessionLabel,
                percent: sess.percent,
                missingCount: sess.missingCount,
                recipientsCount: recipients.length,
              },
            });
          }
        }

        summaries.push({
          tenantId: tenant.id,
          tenantName: tenant.name,
          recalcUpdated: recalc.updated,
          sessionsDetected: incompleteSessions.length,
          notificationsCreated,
          emailsSent,
          emailsDryRun,
        });
      }

      console.log('[qualiopi-blanc-worker] summary', JSON.stringify(summaries, null, 2));
      return summaries;
    }

    export function startQualiopiBlancWorker(): Worker<QualiopiBlancJobPayload> {
      const worker = new Worker<QualiopiBlancJobPayload>(
        QUALIOPI_BLANC_QUEUE_NAME,
        processQualiopiBlancJob as never,
        {
          connection: getWorkerRedis(),
          concurrency: 1,
        },
      );

      worker.on('completed', (job, result) => {
        console.log('[qualiopi-blanc-worker] completed', { jobId: job.id, summaries: result });

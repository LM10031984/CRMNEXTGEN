    ```typescript
    'use server';

    /**
     * Phase 10 — Server action export PDF "Audit Qualiopi blanc" (QBLANC-03).
     *
     * Flow (clone strict Phase 13 `generateVeilleAuditForTheme`) :
     *  1. requireRole(['ADMIN', 'MANAGER']) — LECTEUR ne peut pas déclencher d'export
     *  2. Fetch entries du tenant + score + identité OF
     *  3. Merge avec catalogue (placeholders EN_COURS pour ind sans entry)
     *  4. renderQualiopiBlancAuditHtml → HTML complet
     *  5. renderHtmlToPdfWeasy (footer paged CSS Paged Media)
     *  6. Upload MinIO bucket DOCS_BUCKET (key = `audit-blanc/{tenantId}/{timestamp}-{sha256}.pdf`)
     *  7. prisma.document.create (type='AUDIT_BLANC', entityType='Tenant', entityId=tenantId)
     *  8. logQualiopiBlancEvent(action='qualiopiBlanc.exported', targetEntryId='BULK')
     *  9. revalidatePath('/app/audit-qualiopi-blanc')
     *  10. Retourne { ok, documentId, pdfUrl }
     */

    import { createHash } from 'node:crypto';
    import { revalidatePath } from 'next/cache';
    import { prisma } from '@qualiof/db';
    import { requireRole, UnauthorizedError, ForbiddenError } from '@/lib/rbac';
    import { renderHtmlToPdfWeasy } from '@/lib/pdf-render';
    import { uploadFile, DOCS_BUCKET } from '@/lib/storage';
    import { loadOfConfig } from '@/lib/of-config';
    import {
      renderQualiopiBlancAuditHtml,
      type QualiopiBlancAuditData,
    } from '@/lib/qualiopi-blanc-pdf-template';
    import {
      listEntriesByTenant,
      computeGlobalScore,
    } from '@/lib/qualiopi-blanc/core';
    import { QUALIOPI_INDICATORS_CATALOG } from '@/lib/qualiopi-indicators-catalog';
    import { logQualiopiBlancEvent } from '@/lib/qualiopi-blanc-audit';

    const ROUTE = '/app/audit-qualiopi-blanc';

    export type ExportQualiopiBlancResult =
      | { ok: true; documentId: string; pdfUrl: string }
      | { ok: false; error: string };

    export async function exportQualiopiBlancReport(): Promise<ExportQualiopiBlancResult> {
      try {
        const user = await requireRole(['ADMIN', 'MANAGER']);

        // 1. Fetch data (tenant-scoped)
        const [entries, of] = await Promise.all([
          listEntriesByTenant(user.tenantId),
          loadOfConfig(user.tenantId),
        ]);

        // 2. Merge avec catalogue (placeholders EN_COURS pour ind sans entry)
        const entriesByNumber = new Map(entries.map((e) => [e.indicatorNumber, e]));
        const mergedEntries = QUALIOPI_INDICATORS_CATALOG.map((ind) => {
          const e = entriesByNumber.get(ind.number);
          return {
            indicatorNumber: ind.number,
            status: e?.status ?? 'EN_COURS',
            preuve: e?.preuve ?? null,
            commentaire: e?.commentaire ?? null,
            autoCalculated: e?.autoCalculated ?? false,
            updatedAt: e?.updatedAt ?? null,
          };
        });
        const score = computeGlobalScore(mergedEntries);

        // 3. Build template data
        const data: QualiopiBlancAuditData = {
          tenantName: of.name,
          tenantSiret: of.siret,
          tenantNda: of.rnq,
          generatedAt: new Date(),
          score,
          entries: mergedEntries,
          auditPrevuLabel: 'Samia ZIANI (BCI France) — 03/07/2026',
        };

        // 4. Render HTML
        const html = renderQualiopiBlancAuditHtml(data);

        // 5. Render PDF via WeasyPrint (footer paged CSS Paged Media)
        let pdfBuffer: Buffer;
        try {
          pdfBuffer = await renderHtmlToPdfWeasy(html);
        } catch (e: any) {
          return { ok: false, error: `Erreur rendu PDF audit blanc : ${e?.message ?? e}` };
        }
        if (!pdfBuffer || pdfBuffer.length < 1000) {
          return { ok: false, error: 'PDF audit blanc vide (buffer < 1KB) — vérifier WeasyPrint' };
        }

        // 6. Upload MinIO AVANT INSERT Document (si upload plante, pas d'orphelin BDD)
        const sha256 = createHash('sha256').update(pdfBuffer).digest('hex');
        const timestamp = new Date().toISOString().replace(/[:T]/g, '-').slice(0, 16);
        const objectKey = `audit-blanc/${user.tenantId}/${timestamp}-${sha256.slice(0, 8)}.pdf`;

        try {
          await uploadFile(DOCS_BUCKET, objectKey, pdfBuffer, 'application/pdf');
        } catch (e: any) {
          return { ok: false, error: `Erreur upload MinIO : ${e?.message ?? e}` };
        }

        // 7. INSERT Document (entityType='Tenant', entityId=tenantId — convention dossier tenant-wide)
        const doc = await prisma.document.create({
          data: {
            tenantId: user.tenantId,
            type: 'AUDIT_BLANC',
            entityType: 'Tenant',
            entityId: user.tenantId,
            pdfUrl: objectKey,
            hashSha256: sha256,
          },
        });

        // 8. AuditLog `qualiopiBlanc.exported` (3e verbe sur 4 — convention D-10)
        await logQualiopiBlancEvent({
          tenantId: user.tenantId,
          actorUserId: user.id,
          targetEntryId: 'BULK',
          action: 'qualiopiBlanc.exported',
          diff: {
            documentId: doc.id,
            sha256,
            score: {
              percentConforme: score.percentConforme,
              conforme: score.conforme,
              nonConforme: score.nonConforme,
              enCours: score.enCours,
              na: score.na,
            },
          },
        });

        revalidatePath(ROUTE);
        return { ok: true, documentId: doc.id, pdfUrl: objectKey };
      } catch (e) {
        if (e instanceof UnauthorizedError) return { ok: false, error: 'Non authentifié' };
        if (e instanceof ForbiddenError) return { ok: false, error: 'Accès refusé — ADMIN ou MANAGER requis' };
        console.error('[exportQualiopiBlancReport]', e);
        return { ok: false, error: (e as Error)?.message ?? 'Erreur inconnue' };
      }
    }
    ```

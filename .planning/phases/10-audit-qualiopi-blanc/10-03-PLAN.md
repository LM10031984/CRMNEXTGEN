    ```typescript
    /**
     * Phase 10 — Logique métier "Audit Qualiopi blanc" (worker-safe).
     *
     * CRITICAL — Worker safety (D-03 CONTEXT.md + mémoire `feedback_worker_no_react_imports`) :
     *  - 0 import 'next/cache', 'react', '@/lib/auth', '@/lib/rbac', '@/server/actions'.
     *  - Importable depuis `apps/web/scripts/qualiopi-blanc-worker.ts` (BullMQ daily cron).
     *  - Le wrapper `apps/web/src/server/actions/qualiopi-blanc.ts` ajoute requireRole + revalidatePath.
     *
     * Fonctions exposées :
     *  - calcAutoStatus(indicatorNumber, snapshot)         → calcul statut auto (Ind 9/11/12/30)
     *  - computeGlobalScore(entries)                       → synthèse 4 statuts + %
     *  - upsertEntry(input)                                → CRUD entry (scope tenantId)
     *  - listEntriesByTenant(tenantId)                     → fetch toutes entries d'un tenant
     *  - recalcAutoAllForTenant(tenantId, actorUserId?)    → recalcule les 4 auto-ind
     */

    import { prisma } from '@qualiof/db';
    import type { QualiopiBlancEntry, DocType } from '@qualiof/db';
    import { getById, getBySource } from '../qualiopi-indicators-catalog';
    import { logQualiopiBlancEvent } from '../qualiopi-blanc-audit';

    // ─── Types ───

    export type AutoStatus = 'CONFORME' | 'NON_CONFORME' | 'EN_COURS' | 'NA';

    export interface SessionSnapshot {
      sessionId: string;
      isClosed: boolean; // session.endDate < now
      participantCount: number;
      docsBySession: Set<DocType>; // ex: PROGRAMME, SATISFACTION, CHECKLIST_FORMATION
      participantsWithDocs: Map<string, Set<DocType>>; // participantId → docTypes présents
    }

    export interface GlobalScore {
      total: number;
      conforme: number;
      nonConforme: number;
      enCours: number;
      na: number;
      percentConforme: number; // (conforme + na) / (total - na) * 100, arrondi à l'entier
    }

    // ─── Helpers internes (purs) ───

    /**
     * Ratio de sessions ayant TOUS les DocType requis présents (session-level).
     * Utilisé pour Ind 9 (PROGRAMME, DEROULE_PEDAGOGIQUE) et Ind 30 (SATISFACTION).
     */
    function ratioSessionsWithDocs(
      sessions: SessionSnapshot[],
      requiredDocs: DocType[],
    ): { ratio: number; total: number; ok: number } {
      const closed = sessions.filter((s) => s.isClosed);
      if (closed.length === 0) return { ratio: 1, total: 0, ok: 0 };
      const ok = closed.filter((s) => requiredDocs.every((d) => s.docsBySession.has(d))).length;
      return { ratio: ok / closed.length, total: closed.length, ok };
    }

    /**
     * Ratio de participants ayant TOUS les DocType requis (participant-level).
     * Utilisé pour Ind 11 (EVALUATION_ACQUIS + ATTESTATION_FIN) et Ind 12 (EMARGEMENT + ASSIDUITE).
     */
    function ratioParticipantsWithDocs(
      sessions: SessionSnapshot[],
      requiredDocs: DocType[],
    ): { ratio: number; total: number; ok: number } {
      let total = 0;
      let ok = 0;
      for (const s of sessions) {
        if (!s.isClosed) continue;
        for (const [, docs] of s.participantsWithDocs) {
          total++;
          if (requiredDocs.every((d) => docs.has(d))) ok++;
        }
      }
      if (total === 0) return { ratio: 1, total: 0, ok: 0 };
      return { ratio: ok / total, total, ok };
    }

    // ─── API publique ───

    /**
     * Calcule le statut auto d'un indicateur depuis un snapshot des sessions du tenant.
     *
     * Règles métier (D-02 CONTEXT.md + specifics) :
     *  - Ind 9  : 100% sessions ayant PROGRAMME + DEROULE_PEDAGOGIQUE → CONFORME, sinon EN_COURS
     *  - Ind 11 : 100% participants ayant EVALUATION_ACQUIS + ATTESTATION_FIN → CONFORME, sinon EN_COURS
     *  - Ind 12 : 100% participants ayant EMARGEMENT + ASSIDUITE → CONFORME, sinon EN_COURS
     *  - Ind 30 : >= 90% sessions closes avec SATISFACTION ou SATISFACTION_SESSION → CONFORME, sinon EN_COURS
     *  - Aucune session close (total=0) → EN_COURS (pas de data, pas conclusif)
     *
     * Jette si l'indicateur n'est pas applicable (D-01 figé) ou n'a pas source='auto'.
     */
    export function calcAutoStatus(
      indicatorNumber: number,
      sessions: SessionSnapshot[],
    ): AutoStatus {
      const entry = getById(indicatorNumber);
      if (!entry) throw new Error(`Indicateur ${indicatorNumber} non applicable Start Academy V1`);
      if (entry.source !== 'auto') {
        throw new Error(`Indicateur ${indicatorNumber} a source='manual', pas auto-calculable`);
      }
      if (!entry.dependsOnDocType || entry.dependsOnDocType.length === 0) {
        return 'EN_COURS';
      }

      // Pas de session close → pas de data conclusive
      const closed = sessions.filter((s) => s.isClosed);
      if (closed.length === 0) return 'EN_COURS';

      switch (indicatorNumber) {
        case 9: {
          const { ratio } = ratioSessionsWithDocs(sessions, entry.dependsOnDocType);
          return ratio === 1 ? 'CONFORME' : 'EN_COURS';
        }
        case 11:
        case 12: {
          const { ratio } = ratioParticipantsWithDocs(sessions, entry.dependsOnDocType);
          return ratio === 1 ? 'CONFORME' : 'EN_COURS';
        }
        case 30: {
          // SATISFACTION OU SATISFACTION_SESSION suffit (l'un OU l'autre)
          let ok = 0;
          for (const s of closed) {
            if (s.docsBySession.has('SATISFACTION') || s.docsBySession.has('SATISFACTION_SESSION')) ok++;
          }
          const ratio = ok / closed.length;
          return ratio >= 0.9 ? 'CONFORME' : 'EN_COURS';
        }
        default:
          return 'EN_COURS';
      }
    }

    /**
     * Synthèse globale du tenant : compte par statut + pourcentage conforme.
     *
     * `percentConforme` = (CONFORME + NA) / (total - NA) × 100 — un indicateur NA
     * ne compte ni au num ni au dénom (on évalue 100% des applicables, et NA
     * sort du dénom car non évaluable). Source : audit Qualiopi pratique courante.
     */
    export function computeGlobalScore(
      entries: Array<{ status: AutoStatus | string }>,
    ): GlobalScore {
      const total = entries.length;
      let conforme = 0;
      let nonConforme = 0;
      let enCours = 0;
      let na = 0;
      for (const e of entries) {
        if (e.status === 'CONFORME') conforme++;
        else if (e.status === 'NON_CONFORME') nonConforme++;
        else if (e.status === 'EN_COURS') enCours++;
        else if (e.status === 'NA') na++;
      }
      const denom = total - na;
      const percentConforme = denom > 0 ? Math.round((conforme / denom) * 100) : 0;
      return { total, conforme, nonConforme, enCours, na, percentConforme };
    }

    /**
     * Upsert d'une entry (scope tenantId). Si row existe (couple tenantId+indicatorNumber unique),
     * update les champs fournis ; sinon, crée la row.
     *
     * Retourne l'entry après upsert.
     */
    export async function upsertEntry(input: {
      tenantId: string;
      indicatorNumber: number;
      status: AutoStatus;
      preuve?: string | null;
      preuveDocumentId?: string | null;
      commentaire?: string | null;
      autoCalculated: boolean;
      actorUserId: string | null;
    }): Promise<QualiopiBlancEntry> {
      const entry = getById(input.indicatorNumber);
      if (!entry) {
        throw new Error(
          `Indicateur ${input.indicatorNumber} non applicable Start Academy V1 (D-01)`,
        );
      }

      const result = await prisma.qualiopiBlancEntry.upsert({
        where: {
          tenantId_indicatorNumber: {
            tenantId: input.tenantId,
            indicatorNumber: input.indicatorNumber,
          },
        },
        create: {
          tenantId: input.tenantId,
          indicatorNumber: input.indicatorNumber,
          status: input.status,
          preuve: input.preuve ?? null,
          preuveDocumentId: input.preuveDocumentId ?? null,
          commentaire: input.commentaire ?? null,
          autoCalculated: input.autoCalculated,
          lastUpdatedBy: input.actorUserId,
        },
        update: {
          status: input.status,
          preuve: input.preuve,
          preuveDocumentId: input.preuveDocumentId,
          commentaire: input.commentaire,
          autoCalculated: input.autoCalculated,
          lastUpdatedBy: input.actorUserId,
        },
      });
      return result;
    }

    /** Liste les entries pour un tenant (scope multi-tenant strict). */
    export async function listEntriesByTenant(tenantId: string): Promise<QualiopiBlancEntry[]> {
      return prisma.qualiopiBlancEntry.findMany({
        where: { tenantId },
        orderBy: { indicatorNumber: 'asc' },
      });
    }

    /**
     * Construit le SessionSnapshot[] à partir de Prisma (closed sessions + docs présents).
     *
     * Multi-tenant strict : where tenantId scope partout.
     */
    export async function buildSessionSnapshots(tenantId: string): Promise<SessionSnapshot[]> {
      const sessions = await prisma.trainingSession.findMany({
        where: { tenantId },
        select: {
          id: true,
          endDate: true,
          participants: {
            select: { id: true },
          },
          documents: {
            select: { type: true, participantId: true, entityType: true },
          },
        },
      });
      const now = new Date();
      return sessions.map((s) => {
        const isClosed = s.endDate !== null && s.endDate < now;
        const docsBySession = new Set<DocType>();
        const participantsWithDocs = new Map<string, Set<DocType>>();
        for (const p of s.participants) participantsWithDocs.set(p.id, new Set());
        for (const d of s.documents) {
          if (d.entityType === 'session' && d.participantId === null) {
            docsBySession.add(d.type);
          }
          if (d.entityType === 'participant' && d.participantId) {
            const set = participantsWithDocs.get(d.participantId);
            if (set) set.add(d.type);
          }
        }
        return {
          sessionId: s.id,
          isClosed,
          participantCount: s.participants.length,
          docsBySession,
          participantsWithDocs,
        };
      });
    }

    /**
     * Recalcule TOUS les indicateurs auto (Ind 9, 11, 12, 30) pour 1 tenant et
     * upsert les entries. Trace AuditLog `qualiopiBlanc.auto_recalculated` par
     * indicateur changé.
     *
     * Appelé depuis :
     *  - Worker BullMQ daily (Plan 10-07)
     *  - Server action `recalcAutoStatuses` (Plan 10-03 Task 3)
     *  - Manuel via bouton "Recalculer" sur la page (UI Plan 10-05)
     */
    export async function recalcAutoAllForTenant(
      tenantId: string,
      actorUserId: string | null = null,
    ): Promise<{ updated: number; computed: AutoStatus[] }> {
      const sessions = await buildSessionSnapshots(tenantId);
      const autoIndicators = getBySource('auto'); // 4 entrées : Ind 9, 11, 12, 30
      const computed: AutoStatus[] = [];
      let updated = 0;

      for (const ind of autoIndicators) {
        const newStatus = calcAutoStatus(ind.number, sessions);
        computed.push(newStatus);
        const existing = await prisma.qualiopiBlancEntry.findUnique({
          where: { tenantId_indicatorNumber: { tenantId, indicatorNumber: ind.number } },
        });
        const beforeStatus = existing?.status ?? null;

        // Idempotence : ne touche pas la row si statut inchangé ET autoCalculated=true déjà
        if (existing && existing.status === newStatus && existing.autoCalculated) continue;

        const result = await upsertEntry({
          tenantId,
          indicatorNumber: ind.number,
          status: newStatus,
          autoCalculated: true,
          actorUserId,
        });
        updated++;

        await logQualiopiBlancEvent({
          tenantId,
          actorUserId,
          targetEntryId: result.id,
          action: 'qualiopiBlanc.auto_recalculated',
          diff: {
            indicatorNumber: ind.number,
            status: { before: beforeStatus, after: newStatus },
          },
        });
      }

      return { updated, computed };
    }
    ```

    ```typescript
    import { prisma } from '@qualiof/db';

    const ROUTE = '/app/audit-qualiopi-blanc';

    export type ActionResult<T = unknown> =
      | { ok: true; data?: T }
      | { ok: false; error: string };

    /**
     * Met à jour ou crée une entry (saisie manuelle ADMIN+MANAGER).
     * Trace AuditLog `qualiopiBlanc.entry_updated` avec diff before/after.
     */
    export async function updateEntry(input: UpdateEntryInput): Promise<ActionResult<{ entryId: string }>> {
      try {
        const user = await requireRole(['ADMIN', 'MANAGER']);
        const parsed = UpdateEntrySchema.safeParse(input);
        if (!parsed.success) {
          return { ok: false, error: parsed.error.flatten().formErrors.join(', ') || 'Validation échouée' };
        }

        // Fetch existing pour diff
        const existing = await prisma.qualiopiBlancEntry.findUnique({
          where: {
            tenantId_indicatorNumber: {
              tenantId: user.tenantId,
              indicatorNumber: parsed.data.indicatorNumber,
            },
          },
        });

        const result = await upsertEntry({
          tenantId: user.tenantId,
          indicatorNumber: parsed.data.indicatorNumber,
          status: parsed.data.status,
          preuve: parsed.data.preuve ?? null,
          preuveDocumentId: parsed.data.preuveDocumentId ?? null,
          commentaire: parsed.data.commentaire ?? null,
          autoCalculated: false, // override manuel = autoCalculated désactivé
          actorUserId: user.id,
        });

        await logQualiopiBlancEvent({
          tenantId: user.tenantId,
          actorUserId: user.id,
          targetEntryId: result.id,
          action: 'qualiopiBlanc.entry_updated',
          diff: {
            indicatorNumber: parsed.data.indicatorNumber,
            status: { before: existing?.status ?? null, after: parsed.data.status },
            preuve: { before: existing?.preuve ?? null, after: parsed.data.preuve ?? null },
            commentaire: { before: existing?.commentaire ?? null, after: parsed.data.commentaire ?? null },
          },
        });

        revalidatePath(ROUTE);
        return { ok: true, data: { entryId: result.id } };
      } catch (e) {
        if (e instanceof UnauthorizedError) return { ok: false, error: 'Non authentifié' };
        if (e instanceof ForbiddenError) return { ok: false, error: 'Accès refusé — ADMIN ou MANAGER requis' };
        console.error('[updateEntry]', e);
        return { ok: false, error: (e as Error)?.message ?? 'Erreur inconnue' };
      }
    }

        console.error('[recalcAutoStatuses]', e);
        return { ok: false, error: (e as Error)?.message ?? 'Erreur inconnue' };
      }
    }
    ```

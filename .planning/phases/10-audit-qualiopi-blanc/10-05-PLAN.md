    ```tsx
     * Server Component force-dynamic (data fetchée à chaque requête).

     * RBAC (D-05) : ADMIN + MANAGER + LECTEUR (LECTEUR read-only — pas de bouton

    import { redirect } from 'next/navigation';
    import { Breadcrumb } from '@/components/ui/breadcrumb';
    import { requireRole, hasRole, UnauthorizedError, ForbiddenError } from '@/lib/rbac';
    import { listEntriesByTenant, computeGlobalScore } from '@/lib/qualiopi-blanc/core';
    import { QUALIOPI_INDICATORS_CATALOG } from '@/lib/qualiopi-indicators-catalog';
    import { GlobalScoreCards } from '@/components/audit-qualiopi-blanc/global-score-cards';
    import { IndicatorTable } from '@/components/audit-qualiopi-blanc/indicator-table';
    import { RecalcButton } from '@/components/audit-qualiopi-blanc/recalc-button';

    export const dynamic = 'force-dynamic';

    export default async function AuditQualiopiBlancPage() {
      let user;
      try {
        user = await requireRole(['ADMIN', 'MANAGER', 'LECTEUR']);
      } catch (e) {
        if (e instanceof UnauthorizedError) redirect('/login');
        if (e instanceof ForbiddenError) redirect('/app');
        throw e;
      }

      const canEdit = hasRole(user, ['ADMIN', 'MANAGER']);
      const entries = await listEntriesByTenant(user.tenantId);

      // Construire un map indicatorNumber → entry pour merge avec catalogue
      const entriesByNumber = new Map(entries.map((e) => [e.indicatorNumber, e]));

      // Merge : pour chaque indicateur du catalogue, on a soit l'entry, soit un placeholder EN_COURS
      const rows = QUALIOPI_INDICATORS_CATALOG.map((ind) => {
        const entry = entriesByNumber.get(ind.number);
        return {
          indicator: ind,
          status: entry?.status ?? 'EN_COURS',
          preuve: entry?.preuve ?? null,
          commentaire: entry?.commentaire ?? null,
          preuveDocumentId: entry?.preuveDocumentId ?? null,
          autoCalculated: entry?.autoCalculated ?? false,
          updatedAt: entry?.updatedAt ?? null,
        };
      });

      const score = computeGlobalScore(rows);

      return (
        <div className="space-y-6 p-4 md:p-8">
          <Breadcrumb
            items={[
              { label: 'Tableau de bord', href: '/app' },
              { label: 'Audit Qualiopi blanc' },
            ]}
          />

          <div className="flex flex-wrap items-start justify-between gap-4">
            <div>
              <h1 className="text-2xl font-semibold tracking-tight">Audit Qualiopi blanc</h1>
              <p className="text-sm text-muted-foreground">
                Scoring des 23 indicateurs applicables Start Academy (RNQ V9). Préparation
                audit officiel — outil de pilotage permanent.
              </p>
            </div>
            {canEdit && <RecalcButton />}
          </div>

          <GlobalScoreCards score={score} />

          <IndicatorTable rows={rows} readOnly={!canEdit} />
        </div>
      );
    }
    ```

    ```tsx
    'use client';

    import { useState, useTransition } from 'react';
    import { useForm } from 'react-hook-form';
    import { zodResolver } from '@hookform/resolvers/zod';
    import { toast } from 'sonner';
    import * as Dialog from '@radix-ui/react-dialog';
    import { Pencil, X } from 'lucide-react';
    import { Button } from '@/components/ui/button';
    import { UpdateEntrySchema, type UpdateEntryInput } from '@qualiof/shared/schemas/qualiopi-blanc';
    import { updateEntry } from '@/server/actions/qualiopi-blanc';
    import type { IndicatorRow } from './indicator-table';

    export function IndicatorEditDialog({ row }: { row: IndicatorRow }) {
      const [open, setOpen] = useState(false);
      const [isPending, startTransition] = useTransition();

      const form = useForm<UpdateEntryInput>({
        resolver: zodResolver(UpdateEntrySchema),
        defaultValues: {
          indicatorNumber: row.indicator.number,
          status: row.status as UpdateEntryInput['status'],
          preuve: row.preuve ?? '',
          commentaire: row.commentaire ?? '',
          preuveDocumentId: row.preuveDocumentId ?? null,
        },
      });

      function onSubmit(data: UpdateEntryInput) {
        startTransition(async () => {
          const result = await updateEntry(data);
          if (result.ok) {
            toast.success('Indicateur mis à jour');
            setOpen(false);
          } else {
            toast.error(result.error);
          }
        });
      }

      return (
        <Dialog.Root open={open} onOpenChange={setOpen}>
          <Dialog.Trigger asChild>
            <Button variant="outline" size="sm">
              <Pencil className="h-3 w-3 mr-1" />
              Modifier
            </Button>
          </Dialog.Trigger>
          <Dialog.Portal>
            <Dialog.Overlay className="fixed inset-0 bg-black/50 z-40" />
            <Dialog.Content className="fixed left-1/2 top-1/2 z-50 -translate-x-1/2 -translate-y-1/2 w-full max-w-lg bg-white rounded-lg shadow-lg p-6">
              <div className="flex items-start justify-between mb-4">
                <div>
                  <Dialog.Title className="text-lg font-semibold">
                    Indicateur {row.indicator.number} — {row.indicator.label}
                  </Dialog.Title>
                  <p className="text-xs text-muted-foreground mt-1">
                    Niveau attendu : {row.indicator.niveauAttendu}
                  </p>
                  <p className="text-xs text-red-700 mt-1">
                    Sanction NC : {row.indicator.sanctionNC}
                  </p>
                </div>
                <Dialog.Close className="text-slate-500 hover:text-slate-700">
                  <X className="h-4 w-4" />
                </Dialog.Close>
              </div>

              {row.indicator.source === 'auto' && (
                <p className="text-xs text-amber-700 bg-amber-50 border border-amber-200 rounded p-2 mb-3">
                  ⚠️ Indicateur auto-calculé depuis la matrice docs (Phase 9.1). Modifier
                  manuellement passera <code>autoCalculated=false</code> et désactivera le recalcul.
                </p>
              )}

              <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-3">
                <div>
                  <label className="text-sm font-medium">Statut</label>
                  <select
                    {...form.register('status')}
    ```

    ```typescript
    /**
     * Phase 10 — Template HTML rapport audit Qualiopi blanc (QBLANC-03).
     *
     * Clone-strict pattern Phase 13 `veille-audit-template.ts`. Body sous forme
     * de page de synthèse + 23 pages détaillées (1 par indicateur applicable),
     * page-break-after: always.
     *
     * D-09 Architecture :
     *  1ère page  : En-tête (tenant + SIRET + NDA + date + auditrice prévue) +
     *               synthèse globale (4 stats + score %) + Top 3 risques (sévérité
     *               majeure + status NON_CONFORME).
     *  Pages 2-24 : 1 page par indicateur (libellé, critère, niveau attendu,
     *               sanction NC, verdict actuel, preuve, commentaire, recommandation
     *               si NON_CONFORME).
     *
     * **Moteur PDF : WeasyPrint** (cf feedback_footer_pdf_qualiof.md — Gotenberg
     * footer illisible multi-pages). Footer paged via `renderOfPagedFooter()`
     * (SIRET + NDA + adresse + contact OF).
     *
     * Worker safety : 0 import server-actions/rbac/React/next. Importable depuis
     * action `actions/qualiopi-blanc-export.ts` ET potentiellement depuis script tsx.
     */

    import {
      OF_PAGED_FOOTER_STYLES,
      OF_PAGED_PAGE_RULE,
      renderOfPagedFooter,
    } from './of-paged-footer';
    import {
      QUALIOPI_INDICATORS_CATALOG,
      type QualiopiIndicatorEntry,
    } from './qualiopi-indicators-catalog';
    import type { GlobalScore } from './qualiopi-blanc/core';

    const BRAND_DARK = '#00527A';

    export interface QualiopiBlancEntryView {
      indicatorNumber: number;
      status: 'CONFORME' | 'NON_CONFORME' | 'EN_COURS' | 'NA' | string;
      preuve: string | null;
      commentaire: string | null;
      autoCalculated: boolean;
      updatedAt: Date | null;
    }

    export interface QualiopiBlancAuditData {
      tenantName: string;
      tenantSiret: string;
      tenantNda: string;
      generatedAt: Date;
      score: GlobalScore;
      entries: QualiopiBlancEntryView[]; // 1 par indicateur applicable (ordre indicatorNumber)
      /** Optionnel — auditeur prévu (texte affiché en 1ère page). */
      auditPrevuLabel?: string; // ex: "Samia ZIANI (BCI France) — 03/07/2026"
    }

    function escapeHtml(s: string | null | undefined): string {
      if (s === null || s === undefined) return '';
      return String(s)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }

    function fmtDate(d: Date | null | undefined): string {
      if (!d) return '—';
      return new Intl.DateTimeFormat('fr-FR', {
        day: '2-digit',
        month: '2-digit',
        year: 'numeric',
      }).format(d);
    }

    const STATUS_BADGE: Record<string, { label: string; color: string; bg: string }> = {
      CONFORME: { label: 'Conforme', color: '#065F46', bg: '#D1FAE5' },
      NON_CONFORME: { label: 'Non conforme', color: '#991B1B', bg: '#FEE2E2' },
      EN_COURS: { label: 'En cours', color: '#92400E', bg: '#FEF3C7' },
      NA: { label: 'Non applicable', color: '#374151', bg: '#F3F4F6' },
    };

    function statusBadge(status: string): string {
      const meta = STATUS_BADGE[status] ?? STATUS_BADGE.EN_COURS!;
      return `<span style="display: inline-block; padding: 4px 10px; border-radius: 4px; font-size: 11pt; font-weight: 600; color: ${meta.color}; background: ${meta.bg};">${meta.label}</span>`;
    }

    function renderTop3Risks(
      entries: QualiopiBlancEntryView[],
    ): string {
      const risks = entries
        .filter((e) => e.status === 'NON_CONFORME')
        .map((e) => {
          const ind = QUALIOPI_INDICATORS_CATALOG.find((c) => c.number === e.indicatorNumber);
          if (!ind) return null;
          return { ind, entry: e };
        })
        .filter((x): x is { ind: QualiopiIndicatorEntry; entry: QualiopiBlancEntryView } => x !== null)
        .sort((a, b) => {
          // Sévérité majeure_si_repetee > majeure > mineure
          const sevOrder: Record<string, number> = { majeure_si_repetee: 0, majeure: 1, mineure: 2 };
          return (sevOrder[a.ind.severity] ?? 99) - (sevOrder[b.ind.severity] ?? 99);
        })
        .slice(0, 3);

      if (risks.length === 0) {
        return `<p style="margin-top: 12px; color: #065F46; font-style: italic;">Aucun risque majeur identifié. Bravo !</p>`;
      }

      return `
        <h3 style="margin-top: 16px; font-size: 13pt; color: ${BRAND_DARK};">Top ${risks.length} risque(s) prioritaire(s)</h3>
        <ol style="margin: 8px 0; padding-left: 20px; font-size: 11pt;">
          ${risks
            .map(
              (r) => `<li style="margin-bottom: 8px;">
                <strong>Indicateur ${r.ind.number}</strong> — ${escapeHtml(r.ind.label)} (${r.ind.severity === 'majeure_si_repetee' ? 'NC mineure → majeure' : r.ind.severity === 'majeure' ? 'NC majeure' : 'NC mineure'})
                ${r.ind.recommendationIfNC ? `<br /><span style="color: #4B5563; font-size: 10pt;">Action : ${escapeHtml(r.ind.recommendationIfNC)}</span>` : ''}
              </li>`,
            )
            .join('')}
        </ol>`;
    }

    function renderIndicatorPage(entry: QualiopiBlancEntryView, ind: QualiopiIndicatorEntry): string {
      const recommendation =
        entry.status === 'NON_CONFORME' && ind.recommendationIfNC
          ? `<div style="margin-top: 12px; padding: 12px; background: #FEF3C7; border-left: 4px solid #F59E0B; font-size: 11pt;">
              <strong style="color: #92400E;">Recommandation</strong>
              <p style="margin: 4px 0 0 0;">${escapeHtml(ind.recommendationIfNC)}</p>
            </div>`
          : '';

      return `
        <section class="indicator-page" style="page-break-after: always; padding-top: 8mm;">
          <div style="border-bottom: 2px solid ${BRAND_DARK}; padding-bottom: 8px; margin-bottom: 12px;">
            <p style="margin: 0; font-size: 10pt; color: #6B7280;">Critère ${ind.criterion}</p>
            <h2 style="margin: 4px 0; font-size: 16pt; color: ${BRAND_DARK};">
              Indicateur ${ind.number} — ${escapeHtml(ind.label)}
            </h2>
            <div style="margin-top: 6px;">${statusBadge(entry.status)}
              ${ind.source === 'auto' ? `<span style="margin-left: 8px; font-size: 10pt; color: #6B7280;">(Auto-calculé matrice docs)</span>` : ''}
            </div>
          </div>

          <h3 style="margin-top: 12px; font-size: 12pt; color: #1F2937;">Niveau attendu</h3>
          <p style="font-size: 11pt; margin: 4px 0;">${escapeHtml(ind.niveauAttendu)}</p>

          <h3 style="margin-top: 12px; font-size: 12pt; color: #1F2937;">Sanction en cas de non-conformité</h3>
          <p style="font-size: 11pt; margin: 4px 0; color: #B91C1C;">${escapeHtml(ind.sanctionNC)}</p>

          <h3 style="margin-top: 12px; font-size: 12pt; color: #1F2937;">Preuve présentée</h3>
          <p style="font-size: 11pt; margin: 4px 0; white-space: pre-wrap;">${escapeHtml(entry.preuve) || '<em style="color: #9CA3AF;">Aucune preuve renseignée</em>'}</p>

          <h3 style="margin-top: 12px; font-size: 12pt; color: #1F2937;">Commentaire interne</h3>
          <p style="font-size: 11pt; margin: 4px 0; white-space: pre-wrap;">${escapeHtml(entry.commentaire) || '<em style="color: #9CA3AF;">—</em>'}</p>

          ${recommendation}

          <p style="margin-top: 16px; font-size: 9pt; color: #9CA3AF;">Dernière mise à jour : ${fmtDate(entry.updatedAt)}</p>
        </section>`;
    }

    export function renderQualiopiBlancAuditHtml(data: QualiopiBlancAuditData): string {
      // Pages 2-24 : 1 par indicateur, dans l'ordre du catalogue
      const indicatorPages = QUALIOPI_INDICATORS_CATALOG.map((ind) => {
        const entry = data.entries.find((e) => e.indicatorNumber === ind.number) ?? {
          indicatorNumber: ind.number,
          status: 'EN_COURS',
          preuve: null,
          commentaire: null,
          autoCalculated: false,
          updatedAt: null,
        };
        return renderIndicatorPage(entry, ind);
      }).join('\n');

      return `<!DOCTYPE html>
<html lang="fr">
<head>
  <meta charset="utf-8" />
  <title>Rapport audit Qualiopi blanc — ${escapeHtml(data.tenantName)}</title>
  <style>
    ${OF_PAGED_PAGE_RULE}
    ${OF_PAGED_FOOTER_STYLES}
    body { font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif; color: #1F2937; margin: 0; padding: 0; }
    h1, h2, h3 { margin: 0; padding: 0; }
    .summary-page { page-break-after: always; padding-top: 12mm; }
    .summary-stats { display: flex; gap: 12px; margin: 16px 0; }
    .stat-card { flex: 1; padding: 10px; border: 1px solid #E5E7EB; border-radius: 6px; text-align: center; }
    .stat-value { font-size: 24pt; font-weight: 700; }
    .stat-label { font-size: 10pt; color: #6B7280; text-transform: uppercase; letter-spacing: 0.5px; }
    .tenant-header { border-bottom: 2px solid ${BRAND_DARK}; padding-bottom: 8px; margin-bottom: 12px; }
  </style>
</head>
<body>
  ${renderOfPagedFooter()}

  <!-- Page 1 : Synthèse -->
  <section class="summary-page">
    <div class="tenant-header">
      <h1 style="font-size: 22pt; color: ${BRAND_DARK};">Rapport audit Qualiopi blanc</h1>
      <p style="margin: 6px 0 0 0; font-size: 12pt;">
        <strong>${escapeHtml(data.tenantName)}</strong> · SIRET ${escapeHtml(data.tenantSiret)} · NDA ${escapeHtml(data.tenantNda)}
      </p>
      <p style="margin: 4px 0 0 0; font-size: 10pt; color: #6B7280;">
        Généré le ${fmtDate(data.generatedAt)}${data.auditPrevuLabel ? ` · Audit prévu : ${escapeHtml(data.auditPrevuLabel)}` : ''}
      </p>
    </div>

    <h2 style="font-size: 14pt; color: ${BRAND_DARK};">Synthèse globale</h2>
    <p style="margin: 8px 0; font-size: 13pt;">
      <strong style="font-size: 32pt; color: ${BRAND_DARK};">${data.score.percentConforme}%</strong>
      <span style="margin-left: 8px;">de conformité (${data.score.conforme} / ${data.score.total - data.score.na} indicateurs applicables)</span>
    </p>

    <div class="summary-stats">
      <div class="stat-card" style="border-color: #10B981;">
        <div class="stat-value" style="color: #065F46;">${data.score.conforme}</div>
        <div class="stat-label">Conforme</div>
      </div>
      <div class="stat-card" style="border-color: #EF4444;">
        <div class="stat-value" style="color: #991B1B;">${data.score.nonConforme}</div>
        <div class="stat-label">Non conforme</div>
      </div>
      <div class="stat-card" style="border-color: #F59E0B;">
        <div class="stat-value" style="color: #92400E;">${data.score.enCours}</div>
        <div class="stat-label">En cours</div>
      </div>
      <div class="stat-card" style="border-color: #9CA3AF;">
        <div class="stat-value" style="color: #374151;">${data.score.na}</div>
        <div class="stat-label">N/A</div>
      </div>
    </div>

    ${renderTop3Risks(data.entries)}

    <p style="margin-top: 24px; font-size: 10pt; color: #6B7280; font-style: italic;">
      Ce rapport est un outil de pilotage interne avant audit officiel. Les pages suivantes
      détaillent chaque indicateur applicable (${QUALIOPI_INDICATORS_CATALOG.length} indicateurs) :
      niveau attendu, sanction si non-conforme, verdict actuel, preuve et recommandation.
    </p>
  </section>

  ${indicatorPages}
</body>
</html>`;
    }
    ```

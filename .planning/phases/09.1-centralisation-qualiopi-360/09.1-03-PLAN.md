```typescript
'use client';
import { useState, useEffect, useCallback } from 'react';

/**
 * SSR-safe localStorage state. Phase 9.1 Finding 2.
 *
 * Retourne `initial` jusqu'à hydratation (évite hydration mismatch RSC).
 * Fallback silencieux si localStorage indisponible (private mode, Safari ITP).
 */
export function useLocalStorageState<T>(key: string, initial: T): [T, (next: T) => void, boolean] {
  const [state, setState] = useState<T>(initial);
  const [hydrated, setHydrated] = useState(false);
  useEffect(() => {
    try {
      const raw = window.localStorage.getItem(key);
      if (raw !== null) setState(JSON.parse(raw) as T);
    } catch { /* ignore */ }
    setHydrated(true);
  }, [key]);
  const set = useCallback((next: T) => {
    setState(next);
    try { window.localStorage.setItem(key, JSON.stringify(next)); } catch { /* ignore */ }
  }, [key]);
  return [hydrated ? state : initial, set, hydrated];
}
```

```tsx
'use client';
import Link from 'next/link';
import { cn } from '@/lib/utils';
import { DocStatusBadge } from './doc-status-badge';
import { DocCellMenu } from './doc-cell-menu';
import { DOC_TYPE_LABELS } from '@/lib/doc-scope';
import type { CellState } from '@/lib/derive-cell-state';

interface RowParticipant {
  id: string;
  personId: string;
  fullName: string;        // "Prénom NOM"
  sponsorOrgLabel?: string;
  sponsorOrgId?: string;
}
interface Props {
  participant: RowParticipant;
  cells: Array<{ docType: string; state: CellState }>;
  selected: boolean;
  onSelectionChange: (id: string, next: boolean) => void;
  readOnly: boolean;
  columns: readonly string[]; // ordered docTypes for this matrix instance
}

export function MatrixRow({ participant, cells, selected, onSelectionChange, readOnly, columns }: Props) {
  const cellByType = new Map(cells.map(c => [c.docType, c.state]));
  return (
    <tr
      aria-selected={selected}
      className={cn(
        'hover:bg-muted/30 transition-colors',
        selected && 'border-l-4 border-primary bg-primary-50/30',
      )}
    >
      <td className="sticky left-0 bg-white z-10 px-2 py-1.5 border-b">
        <input
          type="checkbox"
          checked={selected}
          onChange={(e) => onSelectionChange(participant.id, e.target.checked)}
          aria-label={`Sélectionner ${participant.fullName}`}
          disabled={readOnly}
          className="h-4 w-4"
        />
      </td>
      <th scope="row" className="sticky left-[40px] bg-white z-10 px-3 py-1.5 text-left text-sm font-medium border-b min-w-[160px]">
        <Link href={`/app/apprenants/${participant.personId}`} className="hover:text-primary hover:underline truncate block" title={participant.fullName}>
          {participant.fullName}
        </Link>
        {participant.sponsorOrgLabel && (
          <span className="text-xs text-muted-foreground truncate block">{participant.sponsorOrgLabel}</span>
        )}
      </th>
      {columns.map((docType) => {
        const cell = cellByType.get(docType) ?? { state: 'MISSING' as const };
        const label = DOC_TYPE_LABELS[docType]?.long ?? docType;
        return (
          <td key={docType} className="px-1 py-1.5 text-center border-b min-w-[44px]">
            <div className="inline-flex items-center gap-0.5">
              {cell.state === 'GENERATED' && 'pdfRef' in cell && cell.pdfRef ? (
                <a href={`/api/documents/${cell.pdfRef.id}`} target="_blank" rel="noopener" aria-label={`Ouvrir ${label} de ${participant.fullName}`}>
                  <DocStatusBadge state={cell.state} label={label} />
                </a>
              ) : (
                <DocStatusBadge state={cell.state as never} warning={'warning' in cell ? cell.warning : undefined} label={label} />
              )}
              <DocCellMenu
                participantId={participant.id}
                docType={docType}
                state={cell.state as never}
                pdfRef={'pdfRef' in cell ? cell.pdfRef : undefined}
                readOnly={readOnly}
                participantName={participant.fullName}
                docLabel={label}
              />
            </div>
          </td>
        );
      })}
    </tr>
  );
}
```

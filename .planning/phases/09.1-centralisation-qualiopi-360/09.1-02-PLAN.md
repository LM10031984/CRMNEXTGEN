```typescript
'use server';
import { z } from 'zod';
import { revalidatePath } from 'next/cache';
import { prisma } from '@qualiof/db';
import { requireRole, UnauthorizedError, ForbiddenError } from '@/lib/rbac';
import { DocStatusState } from '@qualiof/shared';
import { logDocumentEvent } from '@/lib/document-audit';
import { uploadFile, DOCS_BUCKET } from '@/lib/storage';
import { DOC_TYPE_TO_CLOSURE_KIND } from '@/lib/doc-scope';
import { generateClosurePack } from '@/server/actions/closure-pack';
// Synchronous generators (Convention/AGEFICE/Programme/Convocation) — réutilisés
import { generateConventionForParticipant } from '@/server/actions/...';
import { generateAgeficeForParticipant } from '@/server/actions/...';
import { generateProductProgramme } from '@/server/actions/...';
import crypto from 'node:crypto';

// ============ 1. markDocStatus ============
const MarkDocStatusInputSchema = z.object({
  participantId: z.string().uuid(),
  docType: z.string(),
  state: DocStatusState,
  markedOkWithoutUpload: z.boolean().optional(),
  note: z.string().max(500).optional(),
});

export async function markDocStatus(
  input: z.infer<typeof MarkDocStatusInputSchema>,
): Promise<{ ok: boolean; error?: string }> {
  let user;
  try { user = await requireRole(['ADMIN', 'MANAGER']); }
  catch (e) {
    if (e instanceof UnauthorizedError || e instanceof ForbiddenError) return { ok: false, error: e.message };
    throw e;
  }
  const parsed = MarkDocStatusInputSchema.safeParse(input);
  if (!parsed.success) return { ok: false, error: 'Données invalides' };

  // Tenant scope strict via session.tenantId join
  const participant = await prisma.sessionParticipant.findFirst({
    where: { id: parsed.data.participantId, session: { tenantId: user.tenantId } },
    select: { id: true, sessionId: true, docStatus: true },
  });
  if (!participant) return { ok: false, error: 'Inscription introuvable' };

  const current = (participant.docStatus ?? {}) as Record<string, unknown>;
  const before = current[parsed.data.docType] ?? null;
  const entry = {
    state: parsed.data.state,
    markedOkWithoutUpload: parsed.data.markedOkWithoutUpload,
    note: parsed.data.note,
    updatedAt: new Date().toISOString(),
  };

  // Race condition mitigation (RESEARCH Pitfall 2) : jsonb_set raw + filter tenant strict
  await prisma.$executeRaw`
    UPDATE "SessionParticipant"
    SET "docStatus" = jsonb_set(COALESCE("docStatus", '{}'::jsonb), ${`{${parsed.data.docType}}`}::text[], ${JSON.stringify(entry)}::jsonb, true),
        "updatedAt" = NOW()
    WHERE id = ${parsed.data.participantId}::uuid
      AND "sessionId" IN (SELECT id FROM "TrainingSession" WHERE "tenantId" = ${user.tenantId}::uuid)
  `;

  await logDocumentEvent({
    tenantId: user.tenantId,
    actorUserId: user.id,
    targetEntityId: parsed.data.participantId,
    action: 'documents.status_change',
    diff: { [parsed.data.docType]: { before, after: entry } },
  });

  revalidatePath(`/app/sessions/${participant.sessionId}`);
  return { ok: true };
}

// ============ 2. uploadSignedDoc ============
// Accepte FormData (Next.js Server Action multipart pattern)
export async function uploadSignedDoc(formData: FormData): Promise<{ ok: boolean; error?: string }> {
  let user;
  try { user = await requireRole(['ADMIN', 'MANAGER']); }
  catch (e) {
    if (e instanceof UnauthorizedError || e instanceof ForbiddenError) return { ok: false, error: e.message };
    throw e;
  }
  const participantId = String(formData.get('participantId') ?? '');
  const docType = String(formData.get('docType') ?? '');
  const file = formData.get('file');
  if (!(file instanceof File)) return { ok: false, error: 'Fichier manquant' };
  if (file.type !== 'application/pdf') return { ok: false, error: 'Format non supporté. Le fichier doit être un PDF.' };
  if (file.size > 10 * 1024 * 1024) return { ok: false, error: 'Fichier trop volumineux (max 10 Mo).' };

  const participant = await prisma.sessionParticipant.findFirst({
    where: { id: participantId, session: { tenantId: user.tenantId } },
    select: { id: true, sessionId: true, session: { select: { code: true } }, docStatus: true },
  });
  if (!participant) return { ok: false, error: 'Inscription introuvable' };

  const buf = Buffer.from(await file.arrayBuffer());
  const sha8 = crypto.createHash('sha256').update(buf).digest('hex').slice(0, 8);
  const key = `signed/${user.tenantId}/${participant.session.code}/${participantId}-${docType}-${sha8}.pdf`;
  await uploadFile(DOCS_BUCKET, key, buf, 'application/pdf');

  const current = (participant.docStatus ?? {}) as Record<string, unknown>;
  const before = current[docType] ?? null;
  const entry = {
    state: 'MANUAL_OK' as const,
    uploadedSignedPdfKey: key,
    uploadedSignedAt: new Date().toISOString(),
    uploadedByUserId: user.id,
    updatedAt: new Date().toISOString(),
  };
  await prisma.$executeRaw`
    UPDATE "SessionParticipant"
    SET "docStatus" = jsonb_set(COALESCE("docStatus", '{}'::jsonb), ${`{${docType}}`}::text[], ${JSON.stringify(entry)}::jsonb, true),
        "updatedAt" = NOW()
    WHERE id = ${participantId}::uuid
      AND "sessionId" IN (SELECT id FROM "TrainingSession" WHERE "tenantId" = ${user.tenantId}::uuid)
  `;
  await logDocumentEvent({
    tenantId: user.tenantId,
    actorUserId: user.id,
    targetEntityId: participantId,
    action: 'documents.upload_signed',
    diff: { [docType]: { before, after: { ...entry, uploadedSignedPdfKey: '<key>' } } }, // mask key in log
  });
  revalidatePath(`/app/sessions/${participant.sessionId}`);
  return { ok: true };
}

// ============ 3. regenerateParticipantDoc ============
const RegenerateParticipantDocInputSchema = z.object({
  participantId: z.string().uuid(),
  docKind: z.string(), // MatrixDocType ou ClosureDocKind
});

export async function regenerateParticipantDoc(
  input: z.infer<typeof RegenerateParticipantDocInputSchema>,
): Promise<{ ok: boolean; error?: string; batchId?: string; documentId?: string }> {
  let user;
  try { user = await requireRole(['ADMIN', 'MANAGER']); }
  catch (e) {
    if (e instanceof UnauthorizedError || e instanceof ForbiddenError) return { ok: false, error: e.message };
    throw e;
  }
  const parsed = RegenerateParticipantDocInputSchema.safeParse(input);
  if (!parsed.success) return { ok: false, error: 'Données invalides' };

  const participant = await prisma.sessionParticipant.findFirst({
    where: { id: parsed.data.participantId, session: { tenantId: user.tenantId } },
    select: { id: true, sessionId: true },
  });
  if (!participant) return { ok: false, error: 'Inscription introuvable' };

  await logDocumentEvent({
    tenantId: user.tenantId, actorUserId: user.id, targetEntityId: parsed.data.participantId,
    action: 'documents.regenerate', diff: { docKind: parsed.data.docKind },
  });

  // Routing : synchrone (Convention/AGEFICE/Programme/Convocation) vs BullMQ (ClosureDocKind)
  const closureKind = DOC_TYPE_TO_CLOSURE_KIND[parsed.data.docKind];
  if (closureKind === null) {
    // Synchronous generators — switch case par docKind
    if (parsed.data.docKind === 'CONVENTION') {
      const res = await generateConventionForParticipant(parsed.data.participantId);
      revalidatePath(`/app/sessions/${participant.sessionId}`);
      return res; // { ok, documentId }
    }
    if (parsed.data.docKind === 'AGEFICE') {
      const res = await generateAgeficeForParticipant(parsed.data.participantId);
      revalidatePath(`/app/sessions/${participant.sessionId}`);
      return res;
    }
    if (parsed.data.docKind === 'PROGRAMME') {
      // session-wide doc — délègue au generator produit
      const sess = await prisma.trainingSession.findUnique({ where: { id: participant.sessionId }, select: { productId: true } });
      if (sess?.productId) await generateProductProgramme(sess.productId);
      revalidatePath(`/app/sessions/${participant.sessionId}`);
      return { ok: true };
    }
    return { ok: false, error: `Génération non supportée pour ${parsed.data.docKind}` };
  }

  // ClosureDocKind path → BullMQ avec force=true (RESEARCH Pitfall 1)
  const res = await generateClosurePack(participant.sessionId, {
    participantIds: [parsed.data.participantId],
    kinds: [closureKind as never],
    force: true,
  });
  revalidatePath(`/app/sessions/${participant.sessionId}`);
  return res;
}

// ============ 4. regenerateBatchParticipantDocs ============
const BatchItemSchema = z.object({ participantId: z.string().uuid(), docKind: z.string() });
const RegenerateBatchInputSchema = z.object({
  sessionId: z.string().uuid(),
  items: z.array(BatchItemSchema).min(1).max(100),
});

export async function regenerateBatchParticipantDocs(
  input: z.infer<typeof RegenerateBatchInputSchema>,
): Promise<{ ok: boolean; error?: string; batchId?: string; total?: number }> {
  let user;
  try { user = await requireRole(['ADMIN', 'MANAGER']); }
  catch (e) {
    if (e instanceof UnauthorizedError || e instanceof ForbiddenError) return { ok: false, error: e.message };
    throw e;
  }
  const parsed = RegenerateBatchInputSchema.safeParse(input);
  if (!parsed.success) return { ok: false, error: 'Données invalides' };

  // Group items par closureKind. Filtrer ceux qui sont en mode synchrone (Convention/AGEFICE) — Phase 9.1 limite batch aux ClosureDocKind only (synchrone = bouton ligne uniquement)
  const closureItems = parsed.data.items.filter(i => DOC_TYPE_TO_CLOSURE_KIND[i.docKind]);
  if (closureItems.length === 0) {
    return { ok: false, error: 'Aucun document compatible avec le batch (utilisez le bouton ligne pour Convention/AGEFICE).' };
  }
  // Group par docKind pour 1 generateClosurePack(sessionId, { participantIds, kinds: [kind], force: true }) par kind unique
  const byKind = new Map<string, string[]>();
  for (const it of closureItems) {
    const ck = DOC_TYPE_TO_CLOSURE_KIND[it.docKind]!;
    const arr = byKind.get(ck) ?? [];
    arr.push(it.participantId);
    byKind.set(ck, arr);
  }
  let firstBatchId: string | undefined;
  for (const [kind, pids] of byKind) {
    const res = await generateClosurePack(parsed.data.sessionId, { participantIds: pids, kinds: [kind as never], force: true });
    if (!firstBatchId && res.batchId) firstBatchId = res.batchId;
  }
  await logDocumentEvent({
    tenantId: user.tenantId, actorUserId: user.id, targetEntityId: parsed.data.sessionId,
    action: 'documents.regenerate', diff: { batch: parsed.data.items },
  });
  revalidatePath(`/app/sessions/${parsed.data.sessionId}`);
  return { ok: true, batchId: firstBatchId, total: closureItems.length };
}

// ============ 5. deleteDocument ============
const DeleteDocumentInputSchema = z.object({
  participantId: z.string().uuid(),
  docType: z.string(),
});

export async function deleteDocument(
  input: z.infer<typeof DeleteDocumentInputSchema>,
): Promise<{ ok: boolean; error?: string }> {
  let user;
  try { user = await requireRole(['ADMIN', 'MANAGER']); }
  catch (e) {
    if (e instanceof UnauthorizedError || e instanceof ForbiddenError) return { ok: false, error: e.message };
    throw e;
  }
  const parsed = DeleteDocumentInputSchema.safeParse(input);
  if (!parsed.success) return { ok: false, error: 'Données invalides' };

  const participant = await prisma.sessionParticipant.findFirst({
    where: { id: parsed.data.participantId, session: { tenantId: user.tenantId } },
    select: { id: true, sessionId: true, personId: true, docStatus: true },
  });
  if (!participant) return { ok: false, error: 'Inscription introuvable' };

  await prisma.$transaction(async (tx) => {
    // Delete Document participant-scoped si existe
    await tx.document.deleteMany({ where: { entityType: 'participant', entityId: parsed.data.participantId, type: parsed.data.docType as never } });
    // Reset docStatus[docType] (clé supprimée → MISSING par défaut)
    await tx.$executeRaw`
      UPDATE "SessionParticipant"
      SET "docStatus" = "docStatus" - ${parsed.data.docType}::text,
          "updatedAt" = NOW()
      WHERE id = ${parsed.data.participantId}::uuid
        AND "sessionId" IN (SELECT id FROM "TrainingSession" WHERE "tenantId" = ${user.tenantId}::uuid)
    `;
  });
  await logDocumentEvent({
    tenantId: user.tenantId, actorUserId: user.id, targetEntityId: parsed.data.participantId,
    action: 'documents.delete', diff: { docType: parsed.data.docType },
  });
  revalidatePath(`/app/sessions/${participant.sessionId}`);
  return { ok: true };
}
```

```tsx
import { regenerateParticipantDoc, deleteDocument, markDocStatus } from '@/server/actions/qualiopi-matrix';

interface Props {

  state: 'GENERATED' | 'MANUAL_OK' | 'MISSING' | 'NA';
  pdfRef?: CellPdfRef;
  readOnly?: boolean;
  participantName: string;
  docLabel: string;

export function DocCellMenu({ participantId, docType, state, pdfRef, readOnly, participantName, docLabel }: Props) {
  const [uploadOpen, setUploadOpen] = useState(false);
  const [pending, startTransition] = useTransition();
  const router = useRouter();

  if (state === 'NA') return null;
  if (readOnly) {
    return (
      <button
        type="button"
        disabled
        aria-disabled="true"
        aria-label={`Actions pour ${docLabel} de ${participantName} (lecture seule)`}

      >
        <MoreVertical className="h-4 w-4" aria-hidden />
      </button>
    );

    startTransition(async () => {
      const res = await regenerateParticipantDoc({ participantId, docKind: docType });

        router.refresh();
      } else {
        toast.error(res.error ?? 'Erreur lors de la génération.');
      }
    });

  const handleDelete = () => {
    if (!confirm(`Supprimer ce document ?\nLe PDF généré pour ${participantName} sera définitivement supprimé.`)) return;
    startTransition(async () => {
      const res = await deleteDocument({ participantId, docType });
      if (res.ok) { toast.success('Document supprimé'); router.refresh(); }
      else toast.error(res.error ?? 'Erreur lors de la suppression.');
    });
  };
  const handleMarkNoProof = () => {
    if (!confirm('Marquer signé sans preuve ?\nAucun PDF ne sera téléversé. La pastille orange ⚠ rappellera la dérogation.')) return;
    startTransition(async () => {
      const res = await markDocStatus({ participantId, docType, state: 'MANUAL_OK', markedOkWithoutUpload: true });

    });

  return (
    <>
      <DropdownMenu.Root>
        <DropdownMenu.Trigger asChild>
          <button
            type="button"
            aria-label={`Actions pour ${docLabel} de ${participantName}`}
            aria-haspopup="menu"
            className="p-1 hover:bg-muted/50 rounded focus-visible:ring-2 focus-visible:ring-primary focus-visible:ring-offset-2 focus-visible:outline-none"

            <MoreVertical className="h-4 w-4" aria-hidden />
          </button>
        </DropdownMenu.Trigger>

          <DropdownMenu.Content

            align="end"
            sideOffset={4}
          >
            {state === 'MISSING' && (

                <Sparkles className="h-4 w-4" aria-hidden /> Générer
              </DropdownMenu.Item>
            )}
            {state === 'GENERATED' && (
              <DropdownMenu.Item onSelect={handleRegen} className="flex items-center gap-2 px-3 py-2 text-sm hover:bg-muted cursor-pointer">
                <RefreshCw className="h-4 w-4" aria-hidden /> Re-générer
              </DropdownMenu.Item>
            )}
            {pdfRef && (state === 'GENERATED' || state === 'MANUAL_OK') && (

                <a href={`/api/documents/${pdfRef.id}`} target="_blank" rel="noopener" className="flex items-center gap-2 px-3 py-2 text-sm hover:bg-muted cursor-pointer">
                  <Download className="h-4 w-4" aria-hidden /> Télécharger le PDF
                </a>
              </DropdownMenu.Item>

            <DropdownMenu.Separator className="h-px bg-border my-1" />

              </>
```

```tsx
'use client';
import { useState, useTransition } from 'react';
import * as Dialog from '@radix-ui/react-dialog';
import { Upload, Loader2, X } from 'lucide-react';
import { toast } from 'sonner';
import { useRouter } from 'next/navigation';
import { uploadSignedDoc } from '@/server/actions/qualiopi-matrix';

interface Props {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  participantId: string;
  docType: string;
}

const MAX_BYTES = 10 * 1024 * 1024;

export function UploadSignedDocDialog({ open, onOpenChange, participantId, docType }: Props) {
  const [file, setFile] = useState<File | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [pending, startTransition] = useTransition();
  const router = useRouter();

  const handleFile = (f: File | null) => {
    setError(null);
    if (!f) { setFile(null); return; }
    if (f.type !== 'application/pdf') { setError('Format non supporté. Le fichier doit être un PDF.'); setFile(null); return; }
    if (f.size > MAX_BYTES) { setError('Fichier trop volumineux (max 10 Mo).'); setFile(null); return; }
    setFile(f);
  };

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!file) return;
    const fd = new FormData();
    fd.append('file', file);
    fd.append('participantId', participantId);
    fd.append('docType', docType);
    startTransition(async () => {
      const res = await uploadSignedDoc(fd);
      if (res.ok) {
        toast.success('PDF signé téléversé');
        setFile(null);
        onOpenChange(false);
        router.refresh();
      } else {
        toast.error(res.error ?? 'Erreur lors du téléversement.');
      }
    });
  };

  return (
    <Dialog.Root open={open} onOpenChange={onOpenChange}>
      <Dialog.Portal>
        <Dialog.Overlay className="fixed inset-0 bg-black/40 z-40 data-[state=open]:animate-in data-[state=open]:fade-in-0" />
        <Dialog.Content className="fixed left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 w-[480px] max-w-[90vw] bg-white rounded-lg shadow-xl p-6 z-50">
          <Dialog.Title className="text-lg font-semibold mb-2">Téléverser le PDF signé</Dialog.Title>
          <Dialog.Description className="text-sm text-muted-foreground mb-4">
            Le PDF apparaîtra comme preuve de signature dans la matrice (pastille verte).
          </Dialog.Description>
          <form onSubmit={handleSubmit} className="space-y-4">
            <div>
              <label htmlFor="signed-pdf-file" className="block text-sm font-medium mb-1">Fichier PDF</label>
              <input
                id="signed-pdf-file"
                type="file"
                accept="application/pdf"
                onChange={(e) => handleFile(e.target.files?.[0] ?? null)}
                disabled={pending}
                className="block w-full text-sm focus-visible:ring-2 focus-visible:ring-primary"
              />
              <p className="text-xs text-muted-foreground mt-1">Format PDF · max 10 Mo</p>
              {file && <p className="text-sm text-muted-foreground mt-2">{file.name} · {Math.round(file.size / 1024)} Ko</p>}
              {error && <p role="alert" aria-live="polite" className="text-red-600 text-xs mt-2">{error}</p>}
            </div>
            <div className="flex items-center justify-end gap-2 pt-2">
              <Dialog.Close asChild>
                <button type="button" disabled={pending} className="px-3 py-1.5 text-sm border rounded-md hover:bg-muted">Annuler</button>
              </Dialog.Close>
              <button
                type="submit"
                disabled={!file || pending}
                className="inline-flex items-center gap-2 px-3 py-1.5 text-sm bg-primary text-white rounded-md hover:opacity-90 disabled:opacity-50"
              >
                {pending ? (<><Loader2 className="h-4 w-4 animate-spin" aria-hidden /> Téléversement…</>) : (<><Upload className="h-4 w-4" aria-hidden /> Téléverser</>)}
              </button>
            </div>
          </form>
        </Dialog.Content>
      </Dialog.Portal>
    </Dialog.Root>
  );
}
```

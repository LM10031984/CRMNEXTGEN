```typescript
import { prisma } from '@qualiof/db';

export interface ProductStats {
  sessionsRealisees: number;
  apprenantsFormesTotal: number;
  heuresTotales: number;
  caCumule: number; // en EUR (Decimal -> Number)
  firstYear: number; // pour PrioCard sub "depuis {firstYear}"
}

export async function getProductStats(productId: string, tenantId: string): Promise<ProductStats> {
  const [sessionsCompleted, distinctLearners, sumInvoices, firstSession, product] = await Promise.all([
    prisma.trainingSession.count({
      where: { tenantId, productId, status: 'COMPLETED' },
    }),
    prisma.sessionParticipant.findMany({
      where: { session: { tenantId, productId } },
      select: { personId: true },
      distinct: ['personId'],
    }),
    prisma.invoice.aggregate({
      _sum: { amountHt: true },
      where: {
        tenantId,
        status: 'PAID',
        // Adapter selon schema réel : participant.session.productId OU session.productId direct
        session: { productId },
      } as never,
    }),
    prisma.trainingSession.findFirst({
      where: { tenantId, productId },
      orderBy: { startDate: 'asc' },
      select: { startDate: true },
    }),
    prisma.trainingProduct.findUnique({
      where: { id: productId },
      select: { durationHours: true },
    }),
  ]);

  return {
    sessionsRealisees: sessionsCompleted,
    apprenantsFormesTotal: distinctLearners.length,
    heuresTotales: sessionsCompleted * (product?.durationHours ?? 0),
    caCumule: Number(sumInvoices._sum.amountHt ?? 0),
    firstYear: firstSession?.startDate ? new Date(firstSession.startDate).getFullYear() : new Date().getFullYear(),
  };
}

export async function listProductSessions(productId: string, tenantId: string): Promise<Array<{
  id: string; code: string; startDate: Date; endDate: Date | null; status: string; participantsCount: number;
}>> {
  const sessions = await prisma.trainingSession.findMany({
    where: { tenantId, productId },
    select: {
      id: true, code: true, startDate: true, endDate: true, status: true,
      _count: { select: { participants: true } },
    },
    orderBy: { startDate: 'desc' },
  });
  return sessions.map(s => ({
    id: s.id, code: s.code, startDate: s.startDate, endDate: s.endDate, status: s.status,
    participantsCount: s._count.participants,
  }));
}

export async function listProductLearners(productId: string, tenantId: string): Promise<Array<{
  personId: string; firstName: string; lastName: string; email: string | null; sessionsCount: number; lastSessionDate: Date | null;
}>> {
  // GroupBy personId pour ce produit
  const raw = await prisma.sessionParticipant.groupBy({
    by: ['personId'],
    where: { session: { tenantId, productId } },
    _count: { _all: true },
    _max: { createdAt: true },
  });
  // Hydrate Person data
  const persons = await prisma.person.findMany({
    where: { id: { in: raw.map(r => r.personId) } },
    select: { id: true, firstName: true, lastName: true, email: true },
  });
  const personMap = new Map(persons.map(p => [p.id, p]));
  return raw
    .map(r => {
      const p = personMap.get(r.personId);
      return p
        ? { personId: r.personId, firstName: p.firstName, lastName: p.lastName, email: p.email, sessionsCount: r._count._all, lastSessionDate: r._max.createdAt }
        : null;
    })
    .filter((x): x is NonNullable<typeof x> => x !== null)
    .sort((a, b) => (b.lastSessionDate?.getTime() ?? 0) - (a.lastSessionDate?.getTime() ?? 0));
}
```

```tsx
'use client';
import Link from 'next/link';
import { usePathname, useSearchParams } from 'next/navigation';
import { cn } from '@/lib/utils';

const TABS = [
  { id: 'stats', label: 'Stats' },
  { id: 'sessions', label: 'Sessions' },
  { id: 'apprenants', label: 'Apprenants formés' },
  { id: 'programme', label: 'Programme' },
] as const;

interface Props {
  activeTab: 'stats' | 'sessions' | 'apprenants' | 'programme';
}

export function ProductTabs({ activeTab }: Props) {
  const pathname = usePathname();
  return (
    <nav className="border-b border-border flex overflow-x-auto" role="tablist" aria-label="Onglets fiche produit">
      {TABS.map(tab => {
        const isActive = tab.id === activeTab;
        return (
          <Link
            key={tab.id}
            href={`${pathname}?tab=${tab.id}`}
            role="tab"
            aria-selected={isActive}
            aria-controls={`tab-panel-${tab.id}`}
            id={`tab-${tab.id}`}
            className={cn(
              'px-4 py-3 text-sm whitespace-nowrap border-b-2 transition-colors',
              isActive ? 'border-primary text-primary font-medium' : 'border-transparent text-muted-foreground hover:text-foreground',
            )}
          >
            {tab.label}
          </Link>
        );
      })}
    </nav>
  );
}
```

```tsx
import { Breadcrumb } from '@/components/ui/breadcrumb';
import { ProductTabs } from '@/components/produits/tabs/product-tabs';
import { ProductStatsTab } from '@/components/produits/tabs/product-stats-tab';
import { ProductSessionsTab } from '@/components/produits/tabs/product-sessions-tab';
import { ProductLearnersTab } from '@/components/produits/tabs/product-learners-tab';
import { ProductProgrammeTab } from '@/components/produits/tabs/product-programme-tab';
import { getProductStats, listProductSessions, listProductLearners } from '@/lib/product-stats';

type ValidTab = 'stats' | 'sessions' | 'apprenants' | 'programme';
const VALID_TABS: readonly ValidTab[] = ['stats', 'sessions', 'apprenants', 'programme'] as const;

interface PageProps {
  params: { id: string };
  searchParams?: { tab?: string };
}

export default async function ProductDetailPage({ params, searchParams }: PageProps) {
  const { user } = await validateRequest();
  if (!user) redirect('/login');

  const rawTab = searchParams?.tab;
  const activeTab: ValidTab = (VALID_TABS as readonly string[]).includes(rawTab ?? '')
    ? (rawTab as ValidTab)
    : 'stats';

  const product = await prisma.trainingProduct.findFirst({
    where: { id: params.id, tenantId: user.tenantId },
  });
  if (!product) notFound();

  // Lazy load par tab — seulement les data nécessaires
  let stats, sessions, learners, programmePdf;
  if (activeTab === 'stats') {
    stats = await getProductStats(params.id, user.tenantId);
  } else if (activeTab === 'sessions') {
    sessions = await listProductSessions(params.id, user.tenantId);
  } else if (activeTab === 'apprenants') {
    learners = await listProductLearners(params.id, user.tenantId);
  } else if (activeTab === 'programme') {
    programmePdf = await prisma.document.findFirst({
      where: { tenantId: user.tenantId, entityType: 'product', entityId: params.id, type: 'PROGRAMME' as never },
      select: { id: true },
    });
  }

  return (
```

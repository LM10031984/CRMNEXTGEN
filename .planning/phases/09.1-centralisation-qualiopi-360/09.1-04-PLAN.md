```typescript
import { deriveCellState, type CellState } from './derive-cell-state';
import { MATRIX_DOC_TYPES } from './doc-scope';

export interface ParticipationForStats {
  id: string;
  paymentStatus?: string | null;
  docStatus: Record<string, unknown> | null;
  session: {
    id: string;
    startDate: Date | string;
    product: { durationHours: number; title: string };
  };
}

/**
 * Regroupe les participations par année (basée sur session.startDate).
 *
 * Tri : années desc (2026 avant 2025), dans une année participations desc startDate.
 * Sessions traversant l'année (déc → janv) → assigné à l'année du startDate.
 */
export function groupParticipationsByYear<T extends ParticipationForStats>(
  participations: T[],
): Map<number, T[]> {
  const byYear = new Map<number, T[]>();
  for (const p of participations) {
    const date = typeof p.session.startDate === 'string' ? new Date(p.session.startDate) : p.session.startDate;
    const y = date.getFullYear();
    const arr = byYear.get(y) ?? [];
    arr.push(p);
    byYear.set(y, arr);
  }
  // Tri intra-année desc startDate
  for (const arr of byYear.values()) {
    arr.sort((a, b) => {
      const da = typeof a.session.startDate === 'string' ? new Date(a.session.startDate) : a.session.startDate;
      const db = typeof b.session.startDate === 'string' ? new Date(b.session.startDate) : b.session.startDate;
      return db.getTime() - da.getTime();
    });
  }
  // Tri années desc
  return new Map([...byYear.entries()].sort((a, b) => b[0] - a[0]));
}

/**
 * Somme les `product.durationHours` pour les participations données.
 * Si `year` fourni, filtre uniquement les participations de cette année.
 */
export function computeLearnerHours(
  participations: ParticipationForStats[],
  year?: number,
): number {
  return participations
    .filter((p) => {
      if (year === undefined) return true;
      const d = typeof p.session.startDate === 'string' ? new Date(p.session.startDate) : p.session.startDate;
      return d.getFullYear() === year;
    })
    .reduce((sum, p) => sum + (p.session.product?.durationHours ?? 0), 0);
}

/**
 * Compte les docs Qualiopi manquants + paiements en attente pour l'apprenant.
 *
 * `participantDocsByPid` / `productDocsBySid` / `sessionDocsBySid` / `pedAssetsByPid`
 * sont fournis par le caller (page.tsx) à partir des requêtes Prisma bulk.
 */
export function detectLearnerAlerts(
  participations: ParticipationForStats[],
  productDocsBySid: Map<string, Map<string, { id: string }>>,
  sessionDocsBySid: Map<string, Map<string, { id: string }>>,
  participantDocsByPid: Map<string, Map<string, { id: string }>>,
  pedAssetsByPid: Map<string, Map<string, { id: string }>>,
): { missingDocsCount: number; pendingPaymentsCount: number } {
  let missing = 0;
  let pending = 0;
  for (const p of participations) {
    if (p.paymentStatus === 'PENDING') pending++;
    const participantDocs = participantDocsByPid.get(p.id) ?? new Map();
    const productDocs = productDocsBySid.get(p.session.id) ?? new Map();
    const sessionDocs = sessionDocsBySid.get(p.session.id) ?? new Map();
    const pedAssets = pedAssetsByPid.get(p.id) ?? new Map();
    for (const docType of MATRIX_DOC_TYPES) {
      const cell = deriveCellState(docType, { docStatus: p.docStatus as never }, participantDocs, productDocs, sessionDocs, pedAssets);
      if (cell.state === 'MISSING') missing++;
    }
  }
  return { missingDocsCount: missing, pendingPaymentsCount: pending };
}
```

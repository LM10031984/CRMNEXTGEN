       ```ts
       import type { Prisma } from '@prisma/client';

       /**
        * Génère le prochain numéro de facture séquentiel pour un tenant.
        * Format: {prefix}-NNNNNN (6 chiffres zero-padded).
        * Préfixe lu depuis Tenant.invoicePrefix (fallback 'FAC').
        *
        * USAGE ATOMIQUE: passer le tx d'une prisma.$transaction pour éviter race condition.
        */
       export async function getNextInvoiceNumber(
         tenantId: string,
         tx?: Prisma.TransactionClient,
       ): Promise<string> {
         const db = tx ?? prisma;
         const tenant = await db.tenant.findUnique({
           where: { id: tenantId },
           select: { invoicePrefix: true },
         });
         const prefix = (tenant?.invoicePrefix ?? 'FAC').trim() || 'FAC';
         const last = await db.invoice.findFirst({
           where: { tenantId, number: { startsWith: `${prefix}-` } },
           orderBy: { number: 'desc' },
           select: { number: true },
         });
         const lastNum = last
           ? parseInt(last.number.replace(`${prefix}-`, ''), 10) || 0
           : 0;
         return `${prefix}-${String(lastNum + 1).padStart(6, '0')}`;
       }
       ```

    ```ts
      tenantIdentitySchema,

    type Diff = Record<string, { before: unknown; after: unknown }>;
    type ActionResult =
      | { ok: true }
      | { ok: false; error: string; fieldErrors?: Record<string, string[]> };

    // Helper pur — exporté pour tests
    export function computeDiff<T extends Record<string, unknown>>(before: T, after: T): Diff {
      const diff: Diff = {};
      for (const key of Object.keys(after)) {
        const b = before[key];
        const a = after[key];
        // Comparaison shallow : objets JSON comparés par stringify
        const bStr = b === null || b === undefined ? '' : typeof b === 'object' ? JSON.stringify(b) : String(b);
        const aStr = a === null || a === undefined ? '' : typeof a === 'object' ? JSON.stringify(a) : String(a);
        if (bStr !== aStr) {
          diff[key] = { before: b ?? null, after: a ?? null };
        }
      }
      return diff;
    }

    // Helper — exporté pour tests + Plan 07-03 (upload assets)
    export async function logTenantSettingsChange(opts: {
      tenantId: string;
      userId: string;
      action: string; // 'parameters.update' | 'parameters.upload.logo' | etc.
      diff: Diff;

          userId: opts.userId,

        where: { id: user.tenantId },
    ```
